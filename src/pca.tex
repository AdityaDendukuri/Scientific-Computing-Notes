\subsection{Principal Component Analysis (PCA)}
\label{sub:pca-via-svd}

\textBox{%
\textbf{(Running example: data matrices)}\enspace Least squares used the data matrix to predict a response. PCA uses the same matrix to find the orthogonal directions along which the observations vary most. The SVD answers both questions.
}

\subsubsection{Definitions and Preprocessing}

\addDef{Data Centering}{%
Data matrix $\bX \in \fR^{n\times p}$ ($n$ samples, $p$ features) must be \textbf{centered}:
\begin{align*}
    X_{ij} \leftarrow X_{ij} - \bar{x}_j, \quad \bar{x}_j = \frac{1}{n}\sum_{i=1}^n X_{ij}.
\end{align*}
Throughout this section, $\bX$ is assumed centered.
}
\label{def:pca-centering}

\addDef{Sample Covariance Matrix}{%
The matrix $\bS = \frac{1}{n-1}\bX^T\bX \in \fR^{p\times p}$.
\begin{itemize}
    \item \textbf{Diagonal:}\enspace $S_{jj}$ is the variance of feature $j$.
    \item \textbf{Off-diagonal:}\enspace $S_{ij}$ is the covariance between features $i$ and $j$.
    \item \textbf{Property:}\enspace $\bS$ is symmetric positive semidefinite (SPSD).
\end{itemize}
}
\label{def:sample-covariance}

\addExcr{%
\exDerive
\begin{enumerate}
    \item Show that centering makes each column of $\bX$ have mean zero.
    \item Prove that the diagonal entries of $\bS$ are sample variances.
    \item Prove that $\bS$ is symmetric positive semidefinite by computing $\bz^T\bS\bz$.
\end{enumerate}
}
\label{ex:pca-covariance}

\subsubsection{PCA via SVD}

\addThm{Principal Directions and Scores}{%
Let $\bX = \bU\bsigma\bV^T$ be the SVD of the centered data matrix.
\begin{itemize}
    \item \textbf{Principal Directions (Loadings):}\enspace Columns of $\bV$ (eigenvectors of $\bS$).
    \item \textbf{Principal Components (Scores):}\enspace $\bZ = \bX\bV = \bU\bsigma$.
    \item \textbf{Variances:}\enspace $\lambda_i = \text{Var}(\bz_i) = \frac{\sigma_i^2}{n-1}$.
\end{itemize}
}
\label{thm:pca-svd}

\addPrf{%
Substituting the SVD $\bX = \bU\bsigma\bV^T$ into the sample covariance matrix $\bS$:
\begin{align*}
    \bS &= \frac{1}{n-1} \bX^T\bX \\
    &= \frac{1}{n-1} \left( \bV\bsigma^T\bU^T \right) \left( \bU\bsigma\bV^T \right) \\
    &= \frac{1}{n-1} \bV \bsigma^T (\bU^T\bU) \bsigma \bV^T \\
    &= \bV \left( \frac{\bsigma^2}{n-1} \right) \bV^T.
\end{align*}
Because $\bV$ is orthogonal and $\frac{\bsigma^2}{n-1}$ is diagonal with non-negative entries $\frac{\sigma_i^2}{n-1}$, this is the spectral decomposition of $\bS$. The columns $\bv_i$ are eigenvectors of $\bS$ with eigenvalues $\lambda_i = \sigma_i^2 / (n-1)$.
The principal component score matrix is
\begin{align*}
    \bZ = \bX\bV = \left( \bU\bsigma\bV^T \right) \bV = \bU\bsigma (\bV^T\bV) = \bU\bsigma.
\end{align*}
The sample covariance matrix of the scores is
\begin{align*}
    \bS_{\bZ} = \frac{1}{n-1} \bZ^T\bZ = \frac{1}{n-1} (\bsigma\bU^T)(\bU\bsigma) = \frac{\bsigma^2}{n-1} = \operatorname{diag}(\lambda_1, \dots, \lambda_p),
\end{align*}
confirming that distinct score columns $\bz_i, \bz_j$ have sample covariance zero for $i \neq j$.
}

\addRemark{%
\textbf{(PCA stability)}\enspace Forming $\bX^T\bX$ squares the 2-norm condition number: $\kappa_2(\bX^T\bX)=\kappa_2(\bX)^2$ when $\bX$ has full column rank. Compute the singular vectors directly from an SVD when small singular values or numerical rank matter.
}

\addExcr{%
\exDerive
\begin{enumerate}
    \item Starting from $\bX=\bU\bsigma\bV^T$, compute $\bS=(n-1)^{-1}\bX^T\bX$.
    \item Show that the columns of $\bV$ are eigenvectors of $\bS$.
    \item Show that the corresponding eigenvalues are $\lambda_i=\sigma_i^2/(n-1)$.
    \item Compute the score matrix $\bZ=\bX\bV$ and prove that $\operatorname{Cov}(\bZ)$ is diagonal.
    \item Use Ex~\ref{ex:svd-exercises} to explain the stability warning above.
\end{enumerate}
}
\label{ex:pca-svd}

\subsubsection{Variance and Dimensionality Reduction}

\addDef{Proportion of Variance Explained (PVE)}{%
For the first $k$ components:
\begin{align*}
    \text{PVE}_k = \frac{\sum_{i=1}^k \sigma_i^2}{\sum_{i=1}^p \sigma_i^2}.
\end{align*}
}
\label{def:pve}

\addThm{Optimal Approximation}{%
The rank-$k$ matrix
\begin{align*}
    \bX_k = \sum_{i=1}^k \sigma_i \bu_i \bv_i^T
\end{align*}
is the optimal $k$-dimensional linear approximation of the data in the Frobenius norm, by the Eckart--Young theorem.
}
\label{thm:pca-optimal-approximation}

\addExcr{%
\exDerive
\begin{enumerate}
    \item Use Thm~\ref{thm:eckart-young} to prove Thm~\ref{thm:pca-optimal-approximation}.
    \item Show that $\|\bX-\bX_k\|_F^2=\sum_{i>k}\sigma_i^2$.
    \item Show that $\text{PVE}_k=1-\|\bX-\bX_k\|_F^2/\|\bX\|_F^2$.
    \item Explain why a PVE threshold is a modeling choice, not a theorem.
\end{enumerate}
}
\label{ex:pca-optimal-approximation}

\subsubsection{Implementation Details}

\addRemark{%
\textbf{(Standardization)}\enspace If features have different units, PCA will be dominated by large-scale features. Standardize by dividing each centered column by its standard deviation: $X_{ij} \leftarrow (X_{ij} - \bar{x}_j)/s_j$.
}

\addRemark{%
\textbf{(Interpretation)}\enspace Principal components are directions of large variance. A large loading identifies a direction of variation in the data matrix. It does not by itself explain why that variation occurs.
}

\addDef{Dual PCA for $p \gg n$}{%
When the number of features greatly exceeds the number of samples, we may diagonalize the $n\times n$ Gram matrix $\mathbf{K}=\bX\bX^T$ instead of the $p\times p$ covariance matrix. For each nonzero singular value, the feature-space principal direction is recovered from $\bv_i=\bX^T\bu_i/\sigma_i$. Kernel PCA replaces this linear Gram matrix by a centered nonlinear kernel matrix.
}
\label{def:kernel-pca}

\subsubsection{Exercises}

\addExcr{%
\begin{enumerate}
    \item \exCompute Center $\bX = \begin{pmatrix} 2 & 1 \\ -1 & 3 \\ -1 & -4 \end{pmatrix}$ and compute $\bS$. Find the PVE for $k=1$.
    \item \exDerive Prove that PC scores $\bz_i, \bz_j$ are uncorrelated for $i \neq j$ using Ex~\ref{ex:pca-svd}.
    \item \exCompute Load the Iris dataset. Plot the data in the basis of the first two PCs. Do species cluster?
    \item \exCheck Verify that the sum of variances $\sum \lambda_i$ equals the total variance $\text{Tr}(\bS)$.
\end{enumerate}
}
