\subsection{Principal Component Analysis (PCA)}
\label{sub:pca-via-svd}

\textBox{%
An observation with $p$ measured features is a point in $\fR^p$. A collection of observations may vary strongly along a few directions and very little along the others. Principal component analysis chooses orthogonal directions that retain as much of this variation as possible.
}

\textBox{%
We first subtract the mean observation so that variation is measured about the origin. We then express the variance along a direction as a Rayleigh quotient. The SVD identifies the directions and gives the error caused by retaining only some of them.
}

\subsubsection{Definitions and Preprocessing}

\addDef{Data Centering}{%
Data matrix $\bX \in \fR^{n\times p}$ ($n\geq2$ samples, $p$ features) is first \textbf{centered}:
\begin{align*}
    X_{ij} \leftarrow X_{ij} - \bar{x}_j, \quad \bar{x}_j = \frac{1}{n}\sum_{i=1}^n X_{ij}.
\end{align*}
Throughout this section, $\bX$ is assumed centered, and singular values beyond $\min(n,p)$ are understood to be zero. A variance ratio is defined only when $\|\bX\|_F>0$.
}
\label{def:pca-centering}

\addDef{Sample Covariance Matrix}{%
The matrix $\bS = \frac{1}{n-1}\bX^T\bX \in \fR^{p\times p}$.
\begin{itemize}
    \item \textbf{Diagonal:}\enspace $S_{jj}$ is the variance of feature $j$.
    \item \textbf{Off-diagonal:}\enspace $S_{ij}$ is the covariance between features $i$ and $j$.
    \item \textbf{Property:}\enspace $\bS$ is symmetric positive semidefinite (SPSD).
\end{itemize}
}
\label{def:sample-covariance}

\addExcr{%
\begin{enumerate}
    \item Show that centering makes each column of $\bX$ have mean zero.
    \item Prove that the diagonal entries of $\bS$ are sample variances.
    \item Prove that $\bS$ is symmetric positive semidefinite by computing $\bz^T\bS\bz$.
\end{enumerate}
}
\label{ex:pca-covariance}

\subsubsection{Principal Directions and Scores}

\textBox{%
For a unit direction $\bv\in\fR^p$, the projected observations are the entries of $\bX\bv$. Their mean is zero because the columns of $\bX$ are centered. Their sample variance is therefore
\begin{align*}
 \frac{\|\bX\bv\|_2^2}{n-1}=\bv^T\bS\bv.
\end{align*}
The first principal direction maximizes this variance. By the spectral theorem, it is an eigenvector for the largest eigenvalue of $\bS$. Subsequent directions maximize variance subject to orthogonality to the preceding ones.
}
\addExcr{%
Take centered observations $(2,0)$, $(-2,0)$, $(0,1)$, and $(0,-1)$ as rows of $\bX$. Compute the variance along $\bv=(\cos\theta,\sin\theta)^T$ and find its maximum. Identify the first and second principal directions. What changes if every second coordinate is multiplied by three?
}

\addThm{Principal Directions and Scores}{%
Let $\bX = \bU\bsigma\bV^T$ be the SVD of the centered data matrix.
\begin{itemize}
    \item \textbf{Principal Directions (Loadings):}\enspace Columns of $\bV$ (eigenvectors of $\bS$).
    \item \textbf{Principal Components (Scores):}\enspace $\bZ = \bX\bV = \bU\bsigma$.
    \item \textbf{Variances:}\enspace $\lambda_i = \text{Var}(\bz_i) = \frac{\sigma_i^2}{n-1}$.
\end{itemize}
}
\label{thm:pca-svd}

\addPrf{%
Substituting the SVD $\bX = \bU\bsigma\bV^T$ into the sample covariance matrix $\bS$:
\begin{align*}
    \bS &= \frac{1}{n-1} \bX^T\bX \\
    &= \frac{1}{n-1} \left( \bV\bsigma^T\bU^T \right) \left( \bU\bsigma\bV^T \right) \\
    &= \frac{1}{n-1} \bV \bsigma^T (\bU^T\bU) \bsigma \bV^T \\
    &= \bV \left( \frac{\bsigma^T\bsigma}{n-1} \right) \bV^T.
\end{align*}
Because $\bV$ is orthogonal and $\frac{\bsigma^T\bsigma}{n-1}$ is diagonal with non-negative entries $\frac{\sigma_i^2}{n-1}$, this is the spectral decomposition of $\bS$. The columns $\bv_i$ are eigenvectors of $\bS$ with eigenvalues $\lambda_i = \sigma_i^2 / (n-1)$.
The covariance eigenvalues are therefore squared singular values divided by $n-1$. This division is positive because the sample-size assumption is $n\geq2$. If a singular value is zero, the corresponding direction has zero measured sample variance.

The principal component score matrix is
\begin{align*}
    \bZ = \bX\bV = \left( \bU\bsigma\bV^T \right) \bV = \bU\bsigma (\bV^T\bV) = \bU\bsigma.
\end{align*}
The sample covariance matrix of the scores is
\begin{align*}
    \bS_{\bZ} = \frac{1}{n-1} \bZ^T\bZ = \frac{1}{n-1} (\bsigma^T\bU^T)(\bU\bsigma) = \frac{\bsigma^T\bsigma}{n-1} = \operatorname{diag}(\lambda_1, \dots, \lambda_p),
\end{align*}
confirming that distinct score columns $\bz_i, \bz_j$ have sample covariance zero for $i \neq j$.


Because the centered columns of $\bX$ have zero sums, $\mathbf1^T\bZ=\mathbf1^T\bX\bV=\bzero^T$. Thus the score columns are centered too, making $\bZ^T\bZ/(n-1)$ their sample covariance without an additional centering step. Its diagonal form proves that distinct score columns are uncorrelated. It does not imply statistical independence, which is a stronger claim. See \citet{jolliffe2002} for the statistical interpretation.
}

\addRemark{%
\textbf{(PCA stability)}\enspace Forming $\bX^T\bX$ squares the 2-norm condition number: $\kappa_2(\bX^T\bX)=\kappa_2(\bX)^2$ when $\bX$ has full column rank. Compute the singular vectors directly from an SVD when small singular values or numerical rank matter.
}

\addExcr{%
Show that centering is left multiplication by $\bC=\bI-\mathbf1\mathbf1^T/n$. Prove that $\bC^2=\bC=\bC^T$, deduce $\operatorname{rank}(\bX)\leq n-1$, and explain why at most $n-1$ principal components can have positive variance even when $p>n$.
}
\label{ex:pca-svd}

\subsubsection{Variance and Dimensionality Reduction}

\addDef{Proportion of Variance Explained (PVE)}{%
For the first $k$ components:
\begin{align*}
    \text{PVE}_k = \frac{\sum_{i=1}^k \sigma_i^2}{\sum_{i=1}^p \sigma_i^2}.
\end{align*}
}
\label{def:pve}

\addThm{Optimal Approximation}{%
The rank-$k$ matrix
\begin{align*}
    \bX_k = \sum_{i=1}^k \sigma_i \bu_i \bv_i^T
\end{align*}
is the optimal $k$-dimensional linear approximation of the data in the Frobenius norm, by the Eckart--Young theorem.
}
\label{thm:pca-optimal-approximation}

\addExcr{%
\begin{enumerate}
    \item Use Thm~\ref{thm:eckart-young} to prove Thm~\ref{thm:pca-optimal-approximation}.
    \item Show that $\|\bX-\bX_k\|_F^2=\sum_{i>k}\sigma_i^2$.
    \item Show that $\text{PVE}_k=1-\|\bX-\bX_k\|_F^2/\|\bX\|_F^2$.
    \item Explain why a PVE threshold is a modeling choice, not a theorem.
\end{enumerate}
}
\label{ex:pca-optimal-approximation}

\subsubsection{Implementation Details}

\addRemark{%
\textbf{(Standardization)}\enspace If features have different units, PCA will be dominated by large-scale features. When equal feature weighting is intended, standardize nonconstant columns by their standard deviations: $X_{ij} \leftarrow (X_{ij} - \bar{x}_j)/s_j$.
}

\addRemark{%
\textbf{(Interpretation)}\enspace Principal components are directions of large variance. A large loading identifies a direction of variation in the data matrix. It does not by itself explain why that variation occurs.
}

\addDef{Dual PCA for $p \gg n$}{%
When the number of features greatly exceeds the number of samples, we may diagonalize the $n\times n$ Gram matrix $\mathbf{K}=\bX\bX^T$ instead of the $p\times p$ covariance matrix. For each nonzero singular value, the feature-space principal direction is recovered from $\bv_i=\bX^T\bu_i/\sigma_i$. Kernel PCA replaces this linear Gram matrix by a centered nonlinear kernel matrix.
}
\label{def:kernel-pca}

}
