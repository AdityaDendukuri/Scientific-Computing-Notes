\subsection{Spectral Graph Theory, Diffusion Maps, and Graph Laplacians}
\label{sub:spectral_graph_theory}
\textBox{%
A graph records which pairs of vertices interact and how strongly. A scalar value at each vertex can describe a temperature, a signal, or a quantity attached to an observation. The graph Laplacian compares each value with those of its neighbors.
}

\textBox{%
Its quadratic form adds the squared differences across edges. This connects eigenvalues to connectivity and suggests ways to partition a graph. We then normalize edge weights to obtain a random walk and use its probabilities to compare data points.
}

\subsubsection{Graph Matrices and the Laplacian}

\addDef{Adjacency and Degree Matrices}{%
For a weighted undirected graph $G=(V,E)$ with $n\geq2$ vertices, take $E$ to contain exactly the edges of positive weight:
\begin{itemize}
    \item \textbf{Adjacency $\bA$:}\enspace $A_{ij} = w_{ij} \geq 0$ if $(i,j) \in E$, and $A_{ij} = 0$ otherwise. We assume $w_{ij} = w_{ji}$ and $A_{ii} = 0$.
    \item \textbf{Degree $\bD$:}\enspace Diagonal matrix with entries $D_{ii} = d_i = \sum_{j=1}^n w_{ij}$.
\end{itemize}
}
\label{def:graph-adjacency-degree}

\addExmpl{%
For the path $1$--$2$--$3$ with unit weights, the middle vertex has two neighbors and the endpoints have one. Its Laplacian is
\begin{align*}
 \bL=\begin{pmatrix}1&-1&0\\-1&2&-1\\0&-1&1\end{pmatrix}.
\end{align*}
For values $\bx=(a,b,c)^T$, its quadratic form is $(a-b)^2+(b-c)^2$. A constant signal has zero energy. A jump across either edge has positive energy.
}
\addExcr{%
Derive this quadratic form by multiplying out $\bx^T\bL\bx$. Replace the second edge weight by $\varepsilon>0$ and repeat. Evaluate the form on $(1,1,-2)^T$. How does the energy change as the second edge becomes weak?
}

\addDef{Unnormalized Graph Laplacian}{%
The unnormalized graph Laplacian is
\begin{align*}
    \bL = \bD - \bA.
\end{align*}
For any signal $\bx \in \fR^n$ on the vertices, the action of $\bL$ acts as a discrete Laplace operator:
\begin{align*}
    (\bL \bx)_i = \sum_{j \sim i} w_{ij} (x_i - x_j).
\end{align*}
The associated quadratic form satisfies
\begin{align*}
    \bx^T \bL \bx = \frac{1}{2} \sum_{i,j=1}^n w_{ij} (x_i - x_j)^2 = \sum_{(i,j) \in E} w_{ij} (x_i - x_j)^2.
\end{align*}
}
\label{def:graph-laplacian}

\addProp{Laplacian Positive Semidefinite}{%
For an undirected graph with nonnegative weights, $\bL$ is symmetric positive semidefinite, and $\bL\mathbf{1}=\bzero$.
}
\label{prop:graph-laplacian-psd}

\addPrf{%
Symmetry follows from $w_{ij} = w_{ji}$, so $\bA = \bA^T$ and $\bL = \bD - \bA = \bL^T$.
Expanding the quadratic form algebraically from definitions:
\begin{align*}
    \bx^T \bL \bx &= \bx^T (\bD - \bA) \bx = \sum_{i=1}^n d_i x_i^2 - \sum_{i,j=1}^n w_{ij} x_i x_j \\
    &= \sum_{i=1}^n \left( \sum_{j=1}^n w_{ij} \right) x_i^2 - \sum_{i,j=1}^n w_{ij} x_i x_j \\
    &= \frac{1}{2} \sum_{i,j=1}^n w_{ij} x_i^2 + \frac{1}{2} \sum_{i,j=1}^n w_{ij} x_j^2 - \sum_{i,j=1}^n w_{ij} x_i x_j \\
    &= \frac{1}{2} \sum_{i,j=1}^n w_{ij} (x_i - x_j)^2 = \sum_{(i,j) \in E} w_{ij} (x_i - x_j)^2.
\end{align*}
To obtain the middle equality, interchange the indices in one copy of the degree sum and use $w_{ij}=w_{ji}$. The term $-2x_ix_j$ then combines with $x_i^2+x_j^2$ to form a square. The factor $1/2$ removes the duplicate ordered pairs $(i,j)$ and $(j,i)$.

Because $w_{ij} \geq 0$ and $(x_i - x_j)^2 \geq 0$, each term is non-negative, so $\bx^T \bL \bx \geq 0$ for all $\bx \in \fR^n$, proving positive semidefiniteness.
For $\bx = \mathbf{1}$, every coordinate satisfies $(\bL\mathbf{1})_i = d_i - \sum_{j=1}^n w_{ij} = 0$, so $\bL\mathbf{1} = \bzero$.


The quadratic-form inequality is the definition of positive semidefiniteness. It is not strict positive definiteness, because the nonzero constant vector has zero quadratic form and is annihilated by the matrix.
}

\addThm{Spectral Properties of the Graph Laplacian}{%
Let the eigenvalues of $\bL$ be ordered as $0 = \lambda_1 \leq \lambda_2 \leq \dots \leq \lambda_n$.
\begin{enumerate}
    \item $\lambda_1 = 0$ with eigenvector $\bv_1 = \mathbf{1}$.
    \item The multiplicity of eigenvalue $0$ equals the number of connected components of $G$.
    \item The second smallest eigenvalue $\lambda_2$ is the algebraic connectivity (Fiedler value). It satisfies $\lambda_2 > 0$ if and only if $G$ is connected.
\end{enumerate}
}
\label{thm:graph-spectral-properties}

\addPrf{%
Proposition~\ref{prop:graph-laplacian-psd} shows that all eigenvalues of $\bL$ are nonnegative and that $\mathbf{1}\in\ker(\bL)$, so $\lambda_1=0$.

It remains to identify the entire nullspace. If $\bx\in\ker(\bL)$, then
\begin{align*}
    0=\bx^T\bL\bx=\sum_{(i,j)\in E}w_{ij}(x_i-x_j)^2.
\end{align*}
Every term is nonnegative. Hence $x_i=x_j$ across every edge with positive weight, and therefore $\bx$ is constant on each connected component. Conversely, a vector that is constant on each connected component has zero difference across every edge and belongs to $\ker(\bL)$. A constant on one connected component may be chosen independently of the constants on the other components, because there are no edges between them. The vector equal to one on a given component and zero elsewhere is therefore in the null space.

The indicator vectors of the connected components form a basis for this nullspace, so the multiplicity of eigenvalue zero equals the number of components.

In particular, $G$ is connected exactly when $\ker(\bL)=\operatorname{span}\{\mathbf{1}\}$. In that case zero is a simple eigenvalue, so $\lambda_2>0$. If $G$ is disconnected, the nullspace has dimension at least two and $\lambda_2=0$.


These indicator vectors are independent because their supports are disjoint and nonempty. They span every vector constant on components by assigning its component-wise constants as coefficients. This supplies both halves of the basis assertion and the exact dimension count.
}

\addExcr{%
\textbf{(Path graph Laplacian spectrum)}\enspace For a three-node unweighted path graph $1-2-3$, write down $\bA$ and $\bL$. Compute the eigenvalues and eigenvectors of $\bL$ by hand. Verify that $\lambda_1 = 0$ with $\bv_1 = (1, 1, 1)^T$, and compute the algebraic connectivity $\lambda_2$.
}


\addCor{Rayleigh Characterization of Algebraic Connectivity}{%
For a connected undirected graph,
\begin{align*}
    \lambda_2
    =\min_{\bx\perp\mathbf{1},\,\bx\neq\bzero}
    \frac{\bx^T\bL\bx}{\bx^T\bx}
    =\min_{\bx\perp\mathbf{1},\,\bx\neq\bzero}
    \frac{\sum_{(i,j)\in E}w_{ij}(x_i-x_j)^2}{\sum_i x_i^2}.
\end{align*}
}
\label{cor:graph-algebraic-connectivity}

\addPrf{%
For a connected graph, the null space is spanned by the constant vector. Normalize it as $\bq_1=\mathbf1/\sqrt n$ and complete an orthonormal eigenbasis $\bq_1,\ldots,\bq_n$, with $0=\lambda_1<\lambda_2\leq\cdots\leq\lambda_n$.

Expand a vector perpendicular to $\mathbf1$ in this basis. Its first coefficient is $\bq_1^T\bx=\mathbf1^T\bx/\sqrt n=0$, so $\bx=\sum_{i=2}^nc_i\bq_i$. For a nonzero such vector, at least one remaining coefficient is nonzero. Therefore
\begin{align*}
 \frac{\bx^T\bL\bx}{\bx^T\bx}
 &=\frac{\sum_{i=2}^n\lambda_i c_i^2}{\sum_{i=2}^n c_i^2}\\
 &\geq\frac{\lambda_2\sum_{i=2}^nc_i^2}{\sum_{i=2}^n c_i^2}
 =\lambda_2.
\end{align*}
The lower bound is attained by $\bx=\bq_2$, which is perpendicular to the constant vector. Hence it is the minimum over the entire allowed subspace.

Finally, substitute the graph energy identity $\bx^T\bL\bx=\sum_{(i,j)\in E}w_{ij}(x_i-x_j)^2$ and $\bx^T\bx=\sum_i x_i^2$. This gives the second expression in the statement. The orthogonality constraint removes the constant direction, whose quotient would otherwise give the uninformative minimum zero.
}

\addExcr{%

}


\addDef{Normalized Laplacians}{%
Assume every vertex has positive degree. Normalized Laplacians account for high-degree vertices:
\begin{itemize}
    \item \textbf{Symmetric Normalized Laplacian:}\enspace $\bL_{\text{sym}} = \bD^{-1/2} \bL \bD^{-1/2} = \bI - \bD^{-1/2} \bA \bD^{-1/2}$. Its spectrum lies in the interval $[0, 2]$.
    \item \textbf{Random Walk Laplacian:}\enspace $\bL_{\text{rw}} = \bD^{-1} \bL = \bI - \bD^{-1} \bA$. It is similar to $\bL_{\text{sym}}$ and shares its eigenvalues.
\end{itemize}
}
\label{def:normalized-laplacians}

\addExcr{%
Write $\by=\bD^{-1/2}\bx$ and use $(y_i-y_j)^2\leq2y_i^2+2y_j^2$ to prove $0\leq\bx^T\bL_{\mathrm{sym}}\bx\leq2\|\bx\|_2^2$. Deduce the spectral interval $[0,2]$, and verify the similarity between the two normalized Laplacians. What definition must be added if isolated vertices are allowed?
}


\subsubsection{Spectral Clustering}

\textBox{%
A cut divides the vertices into nonempty sets $A$ and $B$. Let $\operatorname{cut}(A,B)=\sum_{i\in A,j\in B}w_{ij}$ and $\operatorname{vol}(A)=\sum_{i\in A}d_i$. The normalized cut divides the cross-edge weight by the volume on each side and adds the two results:
\begin{align*}
 \operatorname{Ncut}(A,B)
 =\operatorname{cut}(A,B)
 \left(\frac1{\operatorname{vol}(A)}+\frac1{\operatorname{vol}(B)}\right).
\end{align*}
The denominators discourage cutting off a set with tiny total degree. The discrete problem selects vertex sets. A relaxation permits real-valued assignments and gives a lower bound on the best discrete value. Assume positive degrees and a connected graph in the following construction.
}

\addThm{Spectral Relaxation of Graph Cuts}{%
Partitioning a graph into two disjoint sets $A$ and $B$ to minimize the normalized cut is NP-hard. Relaxing the discrete indicator vector to a continuous vector subject to an orthogonality constraint yields the Rayleigh quotient problem
\begin{align*}
    \min_{\bx \perp \bD\mathbf{1},\, \bx \neq \bzero} \frac{\bx^T \bL \bx}{\bx^T \bD \bx}.
\end{align*}
The optimal solution is the Fiedler eigenvector $\bv_2$ of the generalized eigenvalue problem $\bL \bv = \lambda \bD \bv$, or equivalently the second eigenvector of $\bL_{\text{rw}}$. Spectral clustering embeds vertices in $\fR^k$ using the first $k$ nontrivial eigenvectors, then partitions the coordinates using $k$-means.
}
\label{thm:spectral-clustering}

\addExcr{%
For a cut $A,B$, let $v_A=\sum_{i\in A}d_i$ and $v_B=\sum_{i\in B}d_i$. Set $x_i=1/v_A$ on $A$ and $x_i=-1/v_B$ on $B$. Show that $\bx^T\bD\mathbf1=0$ and that its generalized Rayleigh quotient is $\operatorname{cut}(A,B)(1/v_A+1/v_B)$. Then set $\bz=\bD^{1/2}\bx$ and use the spectral theorem to derive the relaxed minimizer stated above. Explain why converting that vector back into a cut is a separate step.
}


\subsubsection{Diffusion Maps and Nonlinear Manifolds}

\addRemark{%
\textbf{(From graphs to manifolds)}\enspace When data points $\bx_1, \dots, \bx_N \in \fR^d$ lie on a low-dimensional manifold embedded in high-dimensional space, ambient Euclidean distances can obscure intrinsic connectivity or create shortcuts across a curved manifold. Diffusion maps define a random walk on a graph constructed over the point cloud and, under suitable sampling and bandwidth limits, approximate diffusion generated by a Laplace--Beltrami operator.
}

\addDef{Diffusion Operator and Random Walk}{%
Given point cloud $\{\bx_i\}_{i=1}^N \subset \fR^d$, construct a pairwise affinity matrix $\bK \in \fR^{N \times N}$ using a Gaussian kernel with bandwidth $\delta > 0$:
\begin{align*}
    K_{ij} = \exp\left(-\frac{\|\bx_i - \bx_j\|^2}{2\delta^2}\right).
\end{align*}
Let $d_i = \sum_{j=1}^N K_{ij}$ denote the row sums, and define $\bD = \operatorname{diag}(d_1, \dots, d_N)$. The row-stochastic diffusion matrix is
\begin{align*}
    \mathbf{M} = \bD^{-1} \bK.
\end{align*}
The entry $M_{ij} = K_{ij}/d_i$ defines the transition probability from point $\bx_i$ to point $\bx_j$ in a single diffusion step.
}
\label{def:diffusion-operator}

\addThm{Spectral Decomposition of the Diffusion Operator}{%
The matrix $\mathbf{M} = \bD^{-1}\bK$ is similar to the real symmetric matrix
\begin{align*}
    \bS = \bD^{-1/2} \bK \bD^{-1/2} = \bD^{1/2} \mathbf{M} \bD^{-1/2}.
\end{align*}
Let $\bS = \bV \boldsymbol{\Lambda} \bV^T$ be the eigendecomposition of $\bS$ with orthonormal eigenvectors $\bV = [\bv_0, \bv_1, \dots, \bv_{N-1}]$ and eigenvalues
\begin{align*}
    1 = \lambda_0 > \lambda_1 \geq \lambda_2 \geq \dots \geq \lambda_{N-1} \geq 0.
\end{align*}
Then $\mathbf{M}$ has real eigenvalues $\lambda_k$ with right eigenvectors $\boldsymbol{\psi}_k = \bD^{-1/2}\bv_k$ and left eigenvectors $\boldsymbol{\phi}_k = \bD^{1/2}\bv_k$. The right eigenvectors satisfy $\mathbf{M} \boldsymbol{\psi}_k = \lambda_k \boldsymbol{\psi}_k$, with $\boldsymbol{\psi}_0=\mathbf1/\sqrt{\sum_i d_i}$ when $\bv_0$ is normalized as above.
}
\label{thm:diffusion-spectral-decomp}

\addPrf{%
All degrees are positive because every Gaussian affinity is positive. Thus the diagonal square roots and inverse square roots of $\bD$ are well defined. Since $\bK$ is symmetric,
\begin{align*}
 \bS^T=(\bD^{-1/2}\bK\bD^{-1/2})^T
 =\bD^{-1/2}\bK\bD^{-1/2}=\bS.
\end{align*}
Also $\bD^{1/2}\mathbf M\bD^{-1/2}=\bS$, so the two matrices are similar.

We justify positive semidefiniteness of the Gaussian kernel rather than inferring it merely from positive entries. Write
\begin{align*}
 K_{ij}=a_i a_j\exp(\bx_i^T\bx_j/\delta^2),
 \qquad a_i=\exp(-\|\bx_i\|_2^2/(2\delta^2)).
\end{align*}
Expand the last exponential in powers. For every nonnegative integer $l$, the kernel $(\bx_i^T\bx_j)^l$ is a Gram kernel: expand the dot product power into sums indexed by $l$ coordinate choices. For real coefficients $c_i$, the resulting quadratic form is a sum of squares of quantities $\sum_i c_i x_{i,s_1}\cdots x_{i,s_l}$. It is nonnegative. The exponential is a convergent sum of these kernels with positive weights $1/(l!\delta^{2l})$. Multiplication by $a_i a_j$ replaces $c_i$ by $a_ic_i$, preserving nonnegativity. Thus $\bK$, and consequently $\bS$, are positive semidefinite.

The symmetric spectral theorem now gives an orthonormal eigenbasis $\bv_k$ of $\bS$ with nonnegative eigenvalues. The row sums of $\mathbf M$ are one. For any eigenvector of $\mathbf M$, selecting an entry of maximal magnitude in its eigenvalue equation gives $|\lambda|\leq1$ by the triangle inequality. Also $\mathbf M\mathbf1=\mathbf1$, so one eigenvalue is one.

Its eigenspace is one-dimensional. For a real vector with $\mathbf M\boldsymbol\psi=\boldsymbol\psi$, a maximal entry is a strictly positive weighted average of all entries. It can equal their maximum only if every entry equals that maximum. Thus the vector is constant. Similarity to a symmetric matrix gives diagonalizability, so the eigenvalue one is algebraically simple as well. Hence all remaining eigenvalues are less than one.

To recover eigenvectors, use $\mathbf M=\bD^{-1/2}\bS\bD^{1/2}$:
\begin{align*}
 \mathbf M(\bD^{-1/2}\bv_k)
 &=\bD^{-1/2}\bS\bv_k
 =\lambda_k\bD^{-1/2}\bv_k,\\
 (\bD^{1/2}\bv_k)^T\mathbf M
 &=\bv_k^T\bS\bD^{1/2}
 =\lambda_k\bv_k^T\bD^{1/2}.
\end{align*}
These are the claimed right and left eigenvectors. Their pairing is $\boldsymbol\phi_j^T\boldsymbol\psi_k=\bv_j^T\bv_k=\delta_{jk}$. Finally the eigenvector of $\bS$ for one is proportional to $\bD^{1/2}\mathbf1$. Its squared norm is $\sum_i d_i$, yielding the stated normalization of $\boldsymbol\psi_0$. Diffusion-map normalization is discussed in \citet{coifmanlafon2006}.
}

\addDef{Diffusion Distance and Diffusion Map Coordinates}{%
The $t$-step diffusion operator $\mathbf{M}^t$ represents running the random walk forward for time $t$. The diffusion distance between points $\bx_i$ and $\bx_j$ measures the difference between their conditional transition distributions weighted by inverse degrees. Here $\pi_i=d_i/\sum_jd_j$ is the stationary probability; using inverse stationary probabilities instead multiplies the following squared distance by $\sum_jd_j$:
\begin{align*}
    D_t(\bx_i, \bx_j)^2
    =
    \|\mathbf{M}^t(i, :) - \mathbf{M}^t(j, :)\|_{\bD^{-1}}^2
    =
    \sum_{k=1}^{N-1} \lambda_k^{2t} \big(\psi_k(i) - \psi_k(j)\big)^2.
\end{align*}
The diffusion map embedding of point $\bx_i$ into an $m$-dimensional Euclidean space is
\begin{align*}
    \Psi_t(\bx_i) = \begin{pmatrix} \lambda_1^t \psi_1(i) \\ \lambda_2^t \psi_2(i) \\ \vdots \\ \lambda_m^t \psi_m(i) \end{pmatrix} \in \fR^m.
\end{align*}
Under this embedding, Euclidean distance in $\fR^m$ equals the truncated diffusion distance:
\begin{align*}
    \|\Psi_t(\bx_i) - \Psi_t(\bx_j)\|_2^2 \approx D_t(\bx_i, \bx_j)^2.
\end{align*}
}
\label{def:diffusion-distance}

\addRemark{%
\textbf{(Diffusion time as a geometric low-pass filter)}\enspace The factor $\lambda_k^{2t}$ in the diffusion distance acts as an exponential low-pass filter on manifold coordinates. Because $\lambda_k < 1$ for all $k \geq 1$, increasing $t$ rapidly suppresses high-frequency modes associated with small eigenvalues, leaving only the dominant macroscopic geometric coordinates.
}

\addRemark{%
\textbf{(Density normalization)}\enspace
The random walk of Def~\ref{def:diffusion-operator} depends on how densely the points sample the manifold as well as on its geometry. Coifman and Lafon separate the two effects by renormalizing the kernel before forming the random walk \citep{coifmanlafon2006}. For a parameter $\alpha\in[0,1]$, they replace $K_{ij}$ by $K_{ij}/(d_i^\alpha d_j^\alpha)$ and then compute new row sums and the row-stochastic matrix as before. In the limit of many points and small bandwidth, $\alpha=0$ gives the classical normalized graph Laplacian, whose limit depends on the sampling density. The choice $\alpha=1/2$ gives the generator of a Fokker--Planck diffusion, and $\alpha=1$ recovers the Laplace--Beltrami operator independently of the density. The renormalized kernel is still symmetric, so Thm~\ref{thm:diffusion-spectral-decomp} applies with the new row sums.
}

\subsubsection{Permutation Invariance and Graph Spectra}

\addRemark{%
\textbf{(Comparing configurations of identical particles)}\enspace In molecular systems and particle clusters with $N$ identical atoms, the state is specified by atomic positions in $\fR^{3N}$. Comparing coordinate vectors requires deciding which atom in one configuration corresponds to which in the other, as well as accounting for rotations and translations. An exhaustive assignment search has $N!$ candidates and becomes expensive quickly. Distance-based graph quantities avoid choosing a coordinate alignment, although a graph spectrum does not uniquely identify every configuration.
}

\addThm{Permutation Invariance of Graph Spectra}{%
Let $\bD \in \fR^{N \times N}$ be the pairwise interatomic distance matrix with entries $D_{ij} = \|\mathbf{r}_i - \mathbf{r}_j\|_2$. Under any atom permutation $\sigma \in S_N$, the distance matrix transforms by an orthogonal permutation matrix $\bP_\sigma$:
\begin{align*}
    \bD_\sigma = \bP_\sigma \bD \bP_\sigma^T.
\end{align*}
Let $\bL$ be any graph Laplacian constructed from edge weights $w_{ij} = f(D_{ij})$. Then the permuted Laplacian is
\begin{align*}
    \bL_\sigma = \bP_\sigma \bL \bP_\sigma^T.
\end{align*}
Because $\bP_\sigma$ is orthogonal, $\bL_\sigma$ is orthogonally similar to $\bL$. Therefore, the eigenvalue spectrum
\begin{align*}
    \operatorname{spec}(\bL) = (\lambda_1, \lambda_2, \dots, \lambda_N)
\end{align*}
is invariant under all atomic permutations $\sigma \in S_N$ and all rigid Euclidean isometries $\mathrm{SE}(3)$.
}
\label{thm:permutation-invariance-spectra}

\addPrf{%
Because rigid rotations and translations preserve pairwise distances $\|\mathbf{r}_i - \mathbf{r}_j\|_2$, the matrix $\bD$ is invariant under $\mathrm{SE}(3)$.

For a rigid transformation $\br_i\mapsto\bR\br_i+\boldsymbol t$ with $\bR^T\bR=\bI$, the difference transforms as $\bR(\br_i-\br_j)$. Its squared length is $(\br_i-\br_j)^T\bR^T\bR(\br_i-\br_j)$, equal to its old squared length. Translation cancels in the difference. Any affinity built only from these distances is therefore unchanged.

Under a permutation $\sigma \in S_N$, vertex $i$ maps to $\sigma(i)$, so the permuted affinity matrix has entries $W_{\sigma(i), \sigma(j)} = W_{ij}$. In matrix notation, $\bW_\sigma = \bP_\sigma \bW \bP_\sigma^T$. Similarly, the degree matrix transforms as $\bD_\sigma = \bP_\sigma \bD \bP_\sigma^T$. Thus
\begin{align*}
    \bL_\sigma = \bD_\sigma - \bW_\sigma = \bP_\sigma (\bD - \bW) \bP_\sigma^T = \bP_\sigma \bL \bP_\sigma^T.
\end{align*}
Because permutation matrices are orthogonal ($\bP_\sigma^{-1} = \bP_\sigma^T$), the characteristic polynomial satisfies
\begin{align*}
    \det(\bL_\sigma - \lambda \bI)
    =
    \det(\bP_\sigma (\bL - \lambda \bI) \bP_\sigma^T)
    =
    \det(\bP_\sigma)\det(\bL - \lambda \bI)\det(\bP_\sigma^T)
    =
    \det(\bL - \lambda \bI).
\end{align*}
The roots of the characteristic polynomial are identical. Thus the eigenvalue multiset $\operatorname{spec}(\bL)$ is invariant under any relabeling of identical particles.


Similarity preserves the multiset of roots, including multiplicity. Sorting these eigenvalues therefore removes the arbitrary vertex ordering. This is an invariance statement, not a uniqueness statement: two different graphs or configurations can still have the same spectrum.
}

\subsubsection{Affinity Scaling and Bandwidth Selection}

\addRemark{%
\textbf{(Sigmoidal behavior of the affinity sum)}\enspace The Gaussian affinity kernel $K_{ij} = \exp(-\|\bx_i - \bx_j\|^2 / (2\delta^2))$ requires selecting the bandwidth parameter $\delta$. If $\delta$ is too small, points are isolated and $K_{ij} \approx 0$ for $i \neq j$. If $\delta$ is too large, points are fully connected and $K_{ij} \approx 1$.

Define the total affinity sum
\begin{align*}
    S(\delta) = \sum_{i=1}^N \sum_{j=1}^N \exp\left(-\frac{\|\bx_i - \bx_j\|^2}{2\delta^2}\right).
\end{align*}
For distinct data points, the log-affinity sum passes between two limiting plateaus. Under adequate sampling of a smooth manifold, an intermediate range may exhibit the scaling below:
\begin{enumerate}
    \item As $\delta \to 0$, $K_{ij} \to 0$ for $i \neq j$, so $S(\delta) \to N$ and $\ln S(\delta) \to \ln N$.
    \item As $\delta \to \infty$, $K_{ij} \to 1$ for all $i,j$, so $S(\delta) \to N^2$ and $\ln S(\delta) \to 2\ln N$.
    \item In an intermediate range where the continuum approximation dominates the diagonal terms, $S(\delta)$ can scale approximately as $\delta^d$, where $d$ is the intrinsic manifold dimension. The slope $\frac{d\ln S(\delta)}{d\ln \delta} \approx d$ estimates dimension.
\end{enumerate}
An intermediate scaling regime is a useful diagnostic for choosing $\delta$, but does not guarantee recovery of manifold geometry. Sampling density, finite sample size, and connectivity must also be checked.
}

\textbf{(Sigmoidal bandwidth selection)}\enspace Let $N = 100$ data points be sampled uniformly from a circle of radius $R = 1$ in $\fR^2$.
    \begin{enumerate}
        \item Write down the formula for the affinity sum $S(\delta) = \sum_{i,j} \exp(-\|\bx_i - \bx_j\|^2 / (2\delta^2))$.
        \item Evaluate the asymptotic limits of $\ln S(\delta)$ as $\delta \to 0$ and $\delta \to \infty$.
        \item For $\delta\in[0.1,0.5]$, estimate the slope numerically. Identify where it is near one and where diagonal contributions or the curvature of the circle affect it. Repeat for two sample sizes and explain why a finite interval of exact scaling is not guaranteed.
    \end{enumerate}
}
