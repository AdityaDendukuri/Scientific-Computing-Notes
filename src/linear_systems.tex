\subsection{Linearity and Fundamental Subspaces}
\label{sub:linearity}

\textBox{%
Linear systems recur throughout scientific computing. Methods for nonlinear equations, optimization, differential equations, and physical simulation routinely reduce a larger problem to a sequence of linear solves. Their reliable solution begins with the existence, uniqueness, and geometric structure encoded by the coefficient matrix.
}

\subsubsection{Linear Systems and Geometric Interpretations}

\textBox{%
A matrix equation $\bA\bx = \bb$ expresses the requirement that a data vector $\bb$ be produced by a linear combination of the columns of $\bA$, weighted by the unknown coordinates $\bx$.
}

\addDef{Linear System}{%
A linear system of $m$ equations in $n$ unknowns is written in matrix-vector form as
\begin{align*}
    \bA\bx = \bb,
\end{align*}
where $\bA \in \fR^{m \times n}$ is the coefficient matrix, $\bx \in \fR^n$ is the unknown solution vector, and $\bb \in \fR^m$ is the right-hand side vector.
}
\label{def:linear-system}

\addRemark{%
\textbf{(Row versus column perspectives)}\enspace
There are two complementary geometric interpretations of $\bA\bx = \bb$:
\begin{enumerate}
    \item \textbf{Row view:}\enspace Each row $\ba_i^T\bx = b_i$ defines an affine hyperplane in $\fR^n$. Solving the system means locating the mutual intersection of these $m$ hyperplanes.
    \item \textbf{Column view:}\enspace Partitioning $\bA = [\ba_1, \dots, \ba_n]$ into its columns, the product evaluates to $\bA\bx = \sum_{j=1}^n x_j \ba_j$. Solving the system means expressing the vector $\bb$ as a linear combination of the column vectors of $\bA$.
\end{enumerate}
}

\addThm{Superposition Principle}{%
Let $\bA \in \fR^{m \times n}$ define a linear operator. If $\bA\bx_1 = \bb_1$ and $\bA\bx_2 = \bb_2$, then for any scalars $\alpha, \beta \in \fR$,
\begin{align*}
    \bA(\alpha\bx_1 + \beta\bx_2) = \alpha\bb_1 + \beta\bb_2.
\end{align*}
Consequently, the solution to a linear system with a composite source term decomposes into a linear combination of individual response vectors.
}
\label{thm:superposition}

\addPrf{%
Using the distributivity and scalar associativity of matrix multiplication,
\begin{align*}
    \bA(\alpha\bx_1 + \beta\bx_2) = \bA(\alpha\bx_1) + \bA(\beta\bx_2) = \alpha(\bA\bx_1) + \beta(\bA\bx_2) = \alpha\bb_1 + \beta\bb_2.
\end{align*}
}

\subsubsection{Invertibility and the Four Fundamental Subspaces}

\textBox{%
The solvability and uniqueness of solutions are governed by four fundamental subspaces associated with $\bA$.
}

\addDef{The Four Fundamental Subspaces}{%
Let $\bA \in \fR^{m \times n}$.
\begin{enumerate}
    \item The \textbf{Column Space (Range)} $\mathcal{R}(\bA) = \{ \bA\bx : \bx \in \fR^n \} \subseteq \fR^m$.
    \item The \textbf{Null Space (Kernel)} $\mathcal{N}(\bA) = \{ \bx \in \fR^n : \bA\bx = \bzero \} \subseteq \fR^n$.
    \item The \textbf{Row Space} $\mathcal{R}(\bA^T) = \{ \bA^T\by : \by \in \fR^m \} \subseteq \fR^n$.
    \item The \textbf{Left Null Space} $\mathcal{N}(\bA^T) = \{ \by \in \fR^m : \bA^T\by = \bzero \} \subseteq \fR^m$.
\end{enumerate}
The rank $r = \operatorname{rank}(\bA)$ is the dimension of the column space $\mathcal{R}(\bA)$.
}
\label{def:four-fundamental-subspaces}

\addThm{The Rank-Nullity Theorem}{%
Let $\bA \in \fR^{m \times n}$ with rank $r = \dim \mathcal{R}(\bA)$. Then
\begin{align*}
    \operatorname{rank}(\bA) + \dim \mathcal{N}(\bA) = n.
\end{align*}
}
\label{thm:rank-nullity}

\addPrf{%
Let $k = \dim \mathcal{N}(\bA)$, and choose a basis $\{\bv_1, \dots, \bv_k\}$ for the null space. Extend this list to a basis $\{\bv_1, \dots, \bv_n\}$ of $\fR^n$.
For any $\bx \in \fR^n$, expand $\bx = \sum_{i=1}^n c_i \bv_i$. Linearity gives
\begin{align*}
    \bA\bx = \sum_{i=1}^k c_i \bA\bv_i + \sum_{i=k+1}^n c_i \bA\bv_i = \bzero + \sum_{i=k+1}^n c_i \bA\bv_i = \sum_{i=k+1}^n c_i \bA\bv_i.
\end{align*}
Thus the vectors $\{\bA\bv_{k+1}, \dots, \bA\bv_n\}$ span $\mathcal{R}(\bA)$.
To establish linear independence, suppose $\sum_{i=k+1}^n d_i \bA\bv_i = \bzero$. Linearity yields
\begin{align*}
    \bA\left(\sum_{i=k+1}^n d_i \bv_i\right) = \bzero \quad \Longrightarrow \quad \bw = \sum_{i=k+1}^n d_i \bv_i \in \mathcal{N}(\bA).
\end{align*}
Because $\{\bv_1, \dots, \bv_k\}$ is a basis for $\mathcal{N}(\bA)$, there exist coefficients $a_1, \dots, a_k$ such that $\bw = \sum_{j=1}^k a_j \bv_j$. Equating the two expressions gives
\begin{align*}
    \sum_{j=1}^k a_j \bv_j - \sum_{i=k+1}^n d_i \bv_i = \bzero.
\end{align*}
Since the full set $\{\bv_1, \dots, \bv_n\}$ is linearly independent, every coefficient vanishes:
\begin{align*}
    a_1 = \dots = a_k = d_{k+1} = \dots = d_n = 0.
\end{align*}
Therefore $\{\bA\bv_{k+1}, \dots, \bA\bv_n\}$ is a basis for $\mathcal{R}(\bA)$, containing exactly $n - k$ vectors. Hence
\begin{align*}
    r = n - k = n - \dim\mathcal{N}(\bA) \quad \Longrightarrow \quad \operatorname{rank}(\bA) + \dim \mathcal{N}(\bA) = n.
\end{align*}
}

\addThm{Fundamental Theorem of Linear Algebra and Orthogonal Decompositions}{%
For any matrix $\bA \in \fR^{m \times n}$:
\begin{enumerate}
    \item The row space and null space are orthogonal complements in $\fR^n$:
    \begin{align*}
        \mathcal{R}(\bA^T) \perp \mathcal{N}(\bA) \qquad \text{and} \qquad \fR^n = \mathcal{R}(\bA^T) \oplus \mathcal{N}(\bA).
    \end{align*}
    \item The column space and left null space are orthogonal complements in $\fR^m$:
    \begin{align*}
        \mathcal{R}(\bA) \perp \mathcal{N}(\bA^T) \qquad \text{and} \qquad \fR^m = \mathcal{R}(\bA) \oplus \mathcal{N}(\bA^T).
    \end{align*}
\end{enumerate}
Consequently, $\dim \mathcal{R}(\bA^T) = \dim \mathcal{R}(\bA) = \operatorname{rank}(\bA)$.
}
\label{thm:fundamental-subspaces-orthogonality}

\addPrf{%
For any $\bx\in\fR^n$, the condition $\bA\bx=\bzero$ says that every row of $\bA$ has zero inner product with $\bx$. Since those rows span $\mathcal{R}(\bA^T)$,
\begin{align*}
    \bx\in\mathcal{N}(\bA)
    \quad\Longleftrightarrow\quad
    \bx\in\mathcal{R}(\bA^T)^\perp.
\end{align*}
Taking orthogonal complements in finite-dimensional space gives
\begin{align*}
    \mathcal{R}(\bA^T)=\mathcal{N}(\bA)^\perp.
\end{align*}
The identical argument with $\bA^T$ gives $\mathcal{R}(\bA)=\mathcal{N}(\bA^T)^\perp$. Each space therefore decomposes into one of these subspaces and its orthogonal complement. Finally, rank--nullity gives
\begin{align*}
    \dim\mathcal{R}(\bA^T)
    =n-\dim\mathcal{N}(\bA)
    =\operatorname{rank}(\bA),
\end{align*}
so row rank equals column rank.
}

\addThm{Solvability and the Left Null Space}{%
The linear system $\bA\bx = \bb$ has a solution if and only if $\bb \in \mathcal{R}(\bA)$.
Equivalently, $\bA\bx = \bb$ is solvable if and only if $\bb$ is orthogonal to every vector in the left null space of $\bA$:
\begin{align*}
    \bA\bx = \bb \text{ has a solution} \iff \by^T\bb = 0 \quad \text{for all } \by \text{ such that } \bA^T\by = \bzero.
\end{align*}
If a solution exists and $\bx_p$ is one particular solution, the complete set of solutions is the affine space
\begin{align*}
    \bx = \bx_p + \bz, \qquad \bz \in \mathcal{N}(\bA).
\end{align*}
}
\label{thm:fredholm-alternative}

\addPrf{%
By Thm~\ref{thm:fundamental-subspaces-orthogonality}, $\fR^m = \mathcal{R}(\bA) \oplus \mathcal{N}(\bA^T)$. Therefore $\bb \in \mathcal{R}(\bA)$ if and only if $\bb \perp \mathcal{N}(\bA^T)$, which means
\begin{align*}
    \langle \by, \bb \rangle = \by^T\bb = 0 \qquad \text{for all } \by \in \mathcal{N}(\bA^T).
\end{align*}
If $\bA\bx_p = \bb$, then for any $\bz \in \mathcal{N}(\bA)$,
\begin{align*}
    \bA(\bx_p + \bz) = \bA\bx_p + \bA\bz = \bb + \bzero = \bb.
\end{align*}
Conversely, if $\bA\bx = \bb$, then
\begin{align*}
    \bA(\bx - \bx_p) = \bA\bx - \bA\bx_p = \bb - \bb = \bzero,
\end{align*}
which implies $\bx - \bx_p \in \mathcal{N}(\bA)$, so $\bx = \bx_p + \bz$ with $\bz \in \mathcal{N}(\bA)$.
}

\addRemark{%
This compatibility condition is often called the finite-dimensional Fredholm alternative. Each left-null vector describes one obstruction that a solvable right-hand side must avoid.
}

\subsubsection{Exercises}

\addExcr{%
\begin{enumerate}
    \item \exCompute \textbf{Dimensions and bases for the four fundamental subspaces.}\enspace Let
    \begin{align*}
        \bA = \begin{pmatrix} 1 & 2 & 1 & 3 \\ 2 & 4 & 1 & 4 \\ 3 & 6 & 2 & 7 \end{pmatrix}.
    \end{align*}
    \begin{enumerate}
        \item Compute the row reduced echelon form of $\bA$. Determine the rank $r = \operatorname{rank}(\bA)$.
        \item Find bases for $\mathcal{R}(\bA)$, $\mathcal{N}(\bA)$, $\mathcal{R}(\bA^T)$, and $\mathcal{N}(\bA^T)$. After studying QR factorization, return and replace them by orthonormal bases.
        \item Confirm that $\dim \mathcal{R}(\bA) + \dim \mathcal{N}(\bA^T) = 3$ and $\dim \mathcal{R}(\bA^T) + \dim \mathcal{N}(\bA) = 4$.
        \item Verify numerically that every vector in your basis of $\mathcal{R}(\bA^T)$ is orthogonal to every vector in your basis of $\mathcal{N}(\bA)$.
    \end{enumerate}
    \item \exDerive \textbf{Decomposition into row space and null space components.}\enspace Let $\bA \in \fR^{m \times n}$.
    \begin{enumerate}
        \item Prove that every vector $\bx \in \fR^n$ has a unique orthogonal decomposition $\bx = \bx_{\text{row}} + \bx_{\text{null}}$ with $\bx_{\text{row}} \in \mathcal{R}(\bA^T)$ and $\bx_{\text{null}} \in \mathcal{N}(\bA)$.
        \item Show that $\bA\bx = \bA\bx_{\text{row}}$, demonstrating that the null-space component contributes nothing to the output.
        \item Prove that among all solutions to a consistent underdetermined system $\bA\bx = \bb$, the unique solution of minimal Euclidean norm $\|\bx\|_2$ lies strictly in the row space: $\bx_{\text{min}} \in \mathcal{R}(\bA^T)$.
    \end{enumerate}
    \item \exDerive \textbf{The Fredholm alternative and inconsistent systems.}\enspace Consider the $3 \times 2$ system
    \begin{align*}
        \bA = \begin{pmatrix} 1 & 1 \\ 1 & 2 \\ 2 & 3 \end{pmatrix}, \qquad \bb = \begin{pmatrix} 1 \\ 2 \\ b_3 \end{pmatrix}.
    \end{align*}
    \begin{enumerate}
        \item Compute a basis for the left null space $\mathcal{N}(\bA^T)$.
        \item Apply Thm~\ref{thm:fredholm-alternative} to find the exact value of $b_3$ that makes the system consistent.
        \item For $b_3 = 5$, find the orthogonal projection of $\bb$ onto $\mathcal{R}(\bA)$ and compute the minimum residual norm $\min_{\bx} \|\bA\bx - \bb\|_2$.
    \end{enumerate}
\end{enumerate}
}
\label{ex:linear-systems-exercises}
