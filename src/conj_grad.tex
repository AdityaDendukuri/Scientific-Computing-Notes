\subsection{The Conjugate Gradient Method}
\label{sub:cg}

\textBox{%
The Conjugate Gradient (CG) method of Hestenes and Stiefel is a standard iterative solver for large, sparse, symmetric positive definite systems. Unlike a stationary iteration determined by one fixed splitting, CG selects the approximation $\bx^{(k)}$ that minimizes the error in the $\bA$-norm over the affine Krylov subspace $\bx^{(0)}+\mathcal{K}_k(\bA,\br^{(0)})$.
}

\textBox{%
Symmetry and conjugacy reduce this expanding-subspace minimization to a short recurrence. The resulting algorithm stores only a few vectors of length $n$ and uses one matrix--vector product per iteration.
}

\textBox{%
\textbf{(Running example: Poisson matrix)}\enspace The discrete Poisson matrix is SPD, so CG applies without changing the model. Its slowly varying eigenmodes explain why the unpreconditioned iteration deteriorates under refinement and why the coarse-grid ideas of Section~\ref{subsec:multigrid} are relevant.
}

\subsubsection{Quadratic Optimization and Steepest Descent}

\textBox{%
The mathematical foundation of the Conjugate Gradient method lies in the equivalence between solving an SPD linear system and minimizing a convex quadratic energy functional.
}

\addThm{Variational Equivalence for SPD Systems}{%
Let $\bA \in \fR^{n \times n}$ be a symmetric positive definite matrix and let $\bb \in \fR^n$. Define the quadratic functional $\phi: \fR^n \to \fR$ by
\begin{align*}
    \phi(\bx) = \frac{1}{2} \bx^T\bA\bx - \bb^T\bx.
\end{align*}
The gradient of $\phi$ is $\nabla \phi(\bx) = \bA\bx - \bb = -\br(\bx)$, where $\br(\bx) = \bb - \bA\bx$ is the residual. Because $\bA$ is positive definite, the Hessian matrix $\nabla^2 \phi(\bx) = \bA$ is strictly positive definite everywhere, and $\phi$ has a unique global minimizer $\bx^*$ satisfying
\begin{align*}
    \nabla \phi(\bx^*) = \bzero \iff \bA\bx^* = \bb.
\end{align*}
Furthermore, the functional error is directly proportional to the squared $\bA$-norm of the solution error:
\begin{align*}
    \phi(\bx) - \phi(\bx^*) = \frac{1}{2} \|\bx - \bx^*\|_\bA^2.
\end{align*}
}
\label{thm:cg-variational}

\addPrf{%
Taylor expanding $\phi(\bx)$ about the exact solution $\bx^* = \bA^{-1}\bb$ gives
\begin{align*}
    \phi(\bx) &= \phi(\bx^*) + \nabla \phi(\bx^*)^T(\bx - \bx^*) + \frac{1}{2} (\bx - \bx^*)^T \bA (\bx - \bx^*) \\
    &= \phi(\bx^*) + \bzero + \frac{1}{2} \|\bx - \bx^*\|_\bA^2.
\end{align*}
Since $\bA$ is positive definite, $\|\bx - \bx^*\|_\bA^2 > 0$ for all $\bx \neq \bx^*$. Thus, minimizing $\phi(\bx)$ over any subspace is strictly equivalent to minimizing the energy norm of the error $\|\bx - \bx^*\|_\bA$.
}

\addRemark{%
\textbf{(The limitation of steepest descent)}\enspace
The method of steepest descent chooses the negative gradient $-\nabla \phi(\bx^{(k)}) = \br^{(k)}$ as its search direction at each step, computing $\bx^{(k+1)} = \bx^{(k)} + \alpha_k \br^{(k)}$ where $\alpha_k$ is the exact line minimizer. Because successive gradients are mutually orthogonal in the Euclidean metric ($\br^{(k+1)T}\br^{(k)} = 0$), the iterates oscillate back and forth along narrow, elongated energy valleys. The convergence rate of steepest descent depends on $(\kappa - 1)/(\kappa + 1)$, leading to severe slowdown when $\kappa_2(\bA) \gg 1$.
}

\subsubsection{Conjugate Directions and Subspace Decoupling}

\textBox{%
To eliminate the inefficient zig-zagging of steepest descent, we replace orthogonal search directions with directions that are orthogonal with respect to the $\bA$-inner product.
}

\addDef{A-Conjugate Vectors}{%
Let $\bA \in \fR^{n \times n}$ be symmetric positive definite. A set of nonzero vectors $\{\bp_0, \bp_1, \dots, \bp_{k-1}\} \subset \fR^n$ is termed \textbf{$\bA$-conjugate} (or $\bA$-orthogonal) if
\begin{align*}
    \bp_i^T \bA \bp_j = 0 \qquad \text{for all } i \neq j.
\end{align*}
Because $\bA$ is positive definite, any set of mutually conjugate nonzero vectors is linearly independent.
}
\label{def:conjugate-directions}

\addThm{Subspace Decoupling and Exact Minimization}{%
Let $\{\bp_0, \dots, \bp_{n-1}\}$ be an $\bA$-conjugate basis for $\fR^n$. The exact solution $\bx^* = \bx^{(0)} + \sum_{i=0}^{n-1} \alpha_i \bp_i$ can be computed by performing $n$ independent one-dimensional line minimizations. The optimal step lengths are given by
\begin{align*}
    \alpha_k = \frac{\bp_k^T \br^{(0)}}{\bp_k^T \bA \bp_k} = \frac{\bp_k^T \br^{(k)}}{\bp_k^T \bA \bp_k}.
\end{align*}
Furthermore, for any $k \leq n$, the partial sum $\bx^{(k)} = \bx^{(0)} + \sum_{i=0}^{k-1} \alpha_i \bp_i$ is the unique vector that minimizes $\|\bx - \bx^*\|_\bA$ over the affine subspace $\bx^{(0)} + \operatorname{span}\{\bp_0, \dots, \bp_{k-1}\}$.
}
\label{thm:conjugate-direction-decoupling}

\addPrf{%
Express the error in the conjugate basis:
\begin{align*}
    \bx - \bx^* = \sum_{i=0}^{n-1} c_i \bp_i.
\end{align*}
Because the directions are $\bA$-conjugate, $\bp_i^T\bA\bp_j = 0$ for $i \neq j$. The squared $\bA$-norm decouples into a sum of scalar terms:
\begin{align*}
    \|\bx - \bx^*\|_\bA^2
    &= \left( \sum_{i=0}^{n-1} c_i \bp_i \right)^T \bA \left( \sum_{j=0}^{n-1} c_j \bp_j \right) \\
    &= \sum_{i=0}^{n-1} \sum_{j=0}^{n-1} c_i c_j \bp_i^T \bA \bp_j
    = \sum_{i=0}^{n-1} c_i^2 \bp_i^T \bA \bp_i.
\end{align*}
Minimizing this sum over the first $k$ coordinates is equivalent to minimizing each term $c_i^2 \bp_i^T \bA \bp_i$ independently.

For the initial guess $\bx^{(0)}$, write $\bx^{(0)} - \bx^* = -\sum_{i=0}^{n-1} \alpha_i \bp_i$. Taking the $\bA$-inner product with $\bp_k$:
\begin{align*}
    \bp_k^T \bA (\bx^{(0)} - \bx^*)
    &= -\sum_{i=0}^{n-1} \alpha_i \bp_k^T \bA \bp_i
    = -\alpha_k \bp_k^T \bA \bp_k.
\end{align*}
Since $\bA\bx^* = \bb$, the left-hand side is $\bp_k^T(\bA\bx^{(0)} - \bb) = -\bp_k^T \br^{(0)}$. Solving for $\alpha_k$:
\begin{align*}
    \alpha_k = \frac{\bp_k^T \br^{(0)}}{\bp_k^T \bA \bp_k}.
\end{align*}
Because $\br^{(k)} = \br^{(0)} - \sum_{i=0}^{k-1} \alpha_i \bA\bp_i$ and $\bp_k^T\bA\bp_i = 0$ for $i < k$, we have $\bp_k^T\br^{(k)} = \bp_k^T\br^{(0)}$, giving $\alpha_k = \frac{\bp_k^T \br^{(k)}}{\bp_k^T \bA \bp_k}$.
}

\subsubsection{The Conjugate Gradient Algorithm}

\textBox{%
The Conjugate Gradient algorithm generates the conjugate directions $\{\bp_k\}$ dynamically from the sequence of residuals $\{\br^{(k)}\}$ via a Gram--Schmidt process. Because $\bA$ is symmetric, all Gram--Schmidt orthogonalization coefficients against previous directions vanish except for the immediately preceding one, yielding a simple two-term recurrence.
}

\addThm{The Conjugate Gradient Algorithm}{%
Let $\bA \in \fR^{n \times n}$ be symmetric positive definite and let $\bb \in \fR^n$. Given an initial estimate $\bx^{(0)}$, compute $\br^{(0)} = \bb - \bA\bx^{(0)}$ and set $\bp^{(0)} = \br^{(0)}$. For $k = 0, 1, 2, \dots$:
\begin{enumerate}
    \item Compute the matrix--vector product $\bw^{(k)} = \bA \bp^{(k)}$.
    \item Compute the step length:
    \begin{align*}
        \alpha_k = \frac{\br^{(k)T}\br^{(k)}}{\bp^{(k)T}\bw^{(k)}}.
    \end{align*}
    \item Update the solution iterate:
    \begin{align*}
        \bx^{(k+1)} = \bx^{(k)} + \alpha_k \bp^{(k)}.
    \end{align*}
    \item Update the residual vector:
    \begin{align*}
        \br^{(k+1)} = \br^{(k)} - \alpha_k \bw^{(k)}.
    \end{align*}
    \item Check termination: if $\|\br^{(k+1)}\|_2 / \|\bb\|_2 \leq \texttt{tol}$, terminate.
    \item Compute the orthogonalization coefficient:
    \begin{align*}
        \beta_k = \frac{\br^{(k+1)T}\br^{(k+1)}}{\br^{(k)T}\br^{(k)}}.
    \end{align*}
    \item Compute the next search direction:
    \begin{align*}
        \bp^{(k+1)} = \br^{(k+1)} + \beta_k \bp^{(k)}.
    \end{align*}
\end{enumerate}
}
\label{thm:cg-algorithm}

\addThm{Orthogonality Properties of CG}{%
In exact arithmetic, the sequences generated by the Conjugate Gradient algorithm satisfy the following orthogonality relations for all $i \neq j$:
\begin{align*}
    \br^{(i)T}\br^{(j)} = 0 \qquad \text{and} \qquad \bp^{(i)T}\bA\bp^{(j)} = 0.
\end{align*}
Moreover, the residual vectors and search directions span the Krylov subspace:
\begin{align*}
    \operatorname{span}\{\br^{(0)}, \br^{(1)}, \dots, \br^{(k)}\} = \operatorname{span}\{\bp^{(0)}, \bp^{(1)}, \dots, \bp^{(k)}\} = \mathcal{K}_{k+1}(\bA, \br^{(0)}).
\end{align*}
Consequently, the method is guaranteed to reach the exact solution in at most $n$ iterations in exact arithmetic.
}
\label{thm:cg-orthogonality}

\addRemark{%
\textbf{(Connection with Lanczos)}\enspace
By Thm~\ref{thm:cg-orthogonality}, the residuals $\br^{(0)},\dots,\br^{(k-1)}$ are mutually orthogonal and the first $j$ of them span $\mathcal{K}_j(\bA,\br^{(0)})$. The Lanczos iteration of Def~\ref{def:lanczos-iteration}, started from $\bv_1=\br^{(0)}/\|\br^{(0)}\|_2$, produces an orthonormal basis with the same nested spans. Such a basis is unique up to the signs of its vectors, and the signs work out to
\begin{align*}
    \bv_{j+1}=(-1)^j\frac{\br^{(j)}}{\|\br^{(j)}\|_2},\qquad j=0,\dots,k-1.
\end{align*}
The CG coefficients also determine the Lanczos matrix $\bT_k$ of Cor~\ref{thm:lanczos-relation}. Its entries are
\begin{align*}
    (\bT_k)_{11}=\frac{1}{\alpha_0},\qquad
    (\bT_k)_{j+1,j+1}=\frac{1}{\alpha_j}+\frac{\beta_{j-1}}{\alpha_{j-1}},\qquad
    (\bT_k)_{j+1,j}=(\bT_k)_{j,j+1}=\frac{\sqrt{\beta_{j-1}}}{\alpha_{j-1}},
\end{align*}
where $\alpha_j$ and $\beta_j$ are the CG step length and direction coefficient. Conjugate gradients may therefore be read as the Lanczos algorithm applied to a linear system.

The correspondence also describes the CG iterate directly. Write $\bx^{(k)}=\bx^{(0)}+\bV_k\by_k$. Since $\br^{(k)}$ is orthogonal to $\mathcal{K}_k$, and since $\bV_k^T\br^{(0)}=\|\br^{(0)}\|_2\be_1$ and $\bV_k^T\bA\bV_k=\bT_k$, we obtain
\begin{align*}
    \bzero=\bV_k^T\br^{(k)}=\bV_k^T(\br^{(0)}-\bA\bV_k\by_k)=\|\br^{(0)}\|_2\be_1-\bT_k\by_k.
\end{align*}
The CG iterate is thus obtained by solving the projected system $\bT_k\by_k=\|\br^{(0)}\|_2\be_1$. In addition, the extreme eigenvalues of $\bT_k$ approach $\lambda_{\min}$ and $\lambda_{\max}$ by Thm~\ref{thm:rayleigh-ritz-optimality}, so the CG coefficients give a running estimate of $\kappa_2(\bA)$ at no extra cost in products with $\bA$.
}

\addExmpl{%
\textbf{(Complete hand trace of a $2 \times 2$ CG solve)}\enspace
Consider solving $\bA\bx = \bb$ with
\begin{align*}
    \bA = \begin{pmatrix} 2 & 1 \\ 1 & 2 \end{pmatrix}, \qquad \bb = \begin{pmatrix} 4 \\ 5 \end{pmatrix}, \qquad \bx^{(0)} = \begin{pmatrix} 0 \\ 0 \end{pmatrix}.
\end{align*}
\begin{enumerate}
    \item \textbf{Initialization ($k=0$):}
    \begin{align*}
        \br^{(0)} = \bb - \bA\bx^{(0)} = \begin{pmatrix} 4 \\ 5 \end{pmatrix}, \qquad \bp^{(0)} = \br^{(0)} = \begin{pmatrix} 4 \\ 5 \end{pmatrix}.
    \end{align*}
    \item \textbf{First Iteration ($k=0$):}
    \begin{align*}
        \bA\bp^{(0)} &= \begin{pmatrix} 2 & 1 \\ 1 & 2 \end{pmatrix} \begin{pmatrix} 4 \\ 5 \end{pmatrix} = \begin{pmatrix} 13 \\ 14 \end{pmatrix}, \\
        \br^{(0)T}\br^{(0)} &= 4^2 + 5^2 = 41, \\
        \bp^{(0)T}\bA\bp^{(0)} &= 4(13) + 5(14) = 52 + 70 = 122, \\
        \alpha_0 &= \frac{41}{122} \approx 0.3361, \\
        \bx^{(1)} &= \bx^{(0)} + \alpha_0 \bp^{(0)} = \frac{41}{122} \begin{pmatrix} 4 \\ 5 \end{pmatrix} = \begin{pmatrix} 164/122 \\ 205/122 \end{pmatrix} \approx \begin{pmatrix} 1.3443 \\ 1.6803 \end{pmatrix}, \\
        \br^{(1)} &= \br^{(0)} - \alpha_0 \bA\bp^{(0)} = \begin{pmatrix} 4 \\ 5 \end{pmatrix} - \frac{41}{122}\begin{pmatrix} 13 \\ 14 \end{pmatrix} = \begin{pmatrix} -45/122 \\ 36/122 \end{pmatrix} \approx \begin{pmatrix} -0.3689 \\ 0.2951 \end{pmatrix}.
    \end{align*}
    The inner product $\br^{(1)T}\br^{(0)} = (-45)(4) + (36)(5) = -180 + 180 = 0$ confirms residual orthogonality.
    \item \textbf{Direction Update ($k=0 \to 1$):}
    \begin{align*}
        \br^{(1)T}\br^{(1)} &= \frac{(-45)^2 + 36^2}{122^2} = \frac{2025 + 1296}{14884} = \frac{3321}{14884}, \\
        \beta_0 &= \frac{\br^{(1)T}\br^{(1)}}{\br^{(0)T}\br^{(0)}} = \frac{3321/14884}{41} = \frac{81}{14884}, \\
        \bp^{(1)} &= \br^{(1)} + \beta_0 \bp^{(0)}
        = \frac{1}{14884}\begin{pmatrix} -5166 \\ 4797 \end{pmatrix}.
    \end{align*}
    Checking conjugacy with $\bA\bp^{(0)}=(13,14)^T$ gives
    \begin{align*}
        \bp^{(1)T}\bA\bp^{(0)}
        =\frac{-5166(13)+4797(14)}{14884}=0.
    \end{align*}
    \item \textbf{Second Iteration ($k=1$):}
    Computing step $\alpha_1$ and updating $\bx^{(2)}$ arrives at the exact solution $\bx^{(2)} = (1, 2)^T$ with residual $\br^{(2)} = \bzero$.
\end{enumerate}
}

\subsubsection{Convergence Rate and Preconditioning}

\textBox{%
Although CG terminates in $n$ steps in exact arithmetic, its primary utility in scientific computing is as a rapid iterative solver for which $k \ll n$.
}

\addThm{Chebyshev Error Bound for Conjugate Gradient}{%
Let $\bA$ be symmetric positive definite with 2-norm condition number $\kappa = \lambda_{\max}/\lambda_{\min}$. The error at step $k$ satisfies
\begin{align*}
    \frac{\|\bx^{(k)} - \bx^*\|_\bA}{\|\bx^{(0)} - \bx^*\|_\bA} \leq 2 \left( \frac{\sqrt{\kappa} - 1}{\sqrt{\kappa} + 1} \right)^k.
\end{align*}
}
\label{thm:cg-convergence-bound}

\addCor{Termination with Distinct Eigenvalues}{%
If $\bA$ has only $m$ distinct eigenvalues, CG reaches the exact solution in at most $m$ iterations in exact arithmetic.
}
\label{cor:cg-distinct-eigenvalues}

\addPrf{%
Let the distinct eigenvalues be $\lambda_1,\dots,\lambda_m$. The polynomial
\begin{align*}
    p_m(t)=\prod_{j=1}^m\left(1-\frac{t}{\lambda_j}\right)
\end{align*}
has degree $m$, satisfies $p_m(0)=1$, and vanishes on the spectrum of $\bA$. The spectral theorem therefore gives $p_m(\bA)=\bzero$. CG minimizes the energy-norm error over all vectors whose error has the form $p(\bA)\be^{(0)}$ with $p(0)=1$. Since $p_m(\bA)\be^{(0)}=\bzero$ is available after $m$ steps, the minimum error is zero.
}

\addRemark{%
\textbf{(The square root condition number advantage)}\enspace
The convergence bound for the Conjugate Gradient method depends on $\sqrt{\kappa}$, whereas stationary iterations depend directly on $\kappa$. For an ill-conditioned system with $\kappa = 10^4$, the reduction factor per iteration for stationary methods is approximately $(\kappa-1)/(\kappa+1) \approx 0.9998$, whereas for CG it is $(\sqrt{\kappa}-1)/(\sqrt{\kappa}+1) = 99/101 \approx 0.9802$, converging fifty times faster per step.
}

\addRemark{%
\textbf{(Superlinear convergence)}\enspace
The Chebyshev bound is a worst-case guarantee for an arbitrary spectrum. When the eigenvalues of $\bA$ are grouped into tight clusters with only a few isolated outliers, CG eliminates the outlying eigenmodes in the first few iterations and then converges at a rate governed by the clustered eigenvalues. This behavior is termed superlinear convergence.
}

\addDef{Preconditioned Conjugate Gradient (PCG)}{%
Let $\bM$ be a symmetric positive definite preconditioner. The Preconditioned Conjugate Gradient algorithm solves $\bA\bx = \bb$ by transforming the geometry using the inner product induced by $\bM^{-1}$. At each step, a linear system $\bM\bz^{(k)} = \br^{(k)}$ is solved, and the step lengths and direction updates become
\begin{align*}
    \alpha_k = \frac{\br^{(k)T}\bz^{(k)}}{\bp^{(k)T}\bA\bp^{(k)}}, \qquad \beta_k = \frac{\br^{(k+1)T}\bz^{(k+1)}}{\br^{(k)T}\bz^{(k)}}.
\end{align*}
A well-chosen preconditioner clusters the eigenvalues of $\bM^{-1}\bA$ near $1$, ensuring rapid convergence in few iterations.
}
\label{def:pcg}
