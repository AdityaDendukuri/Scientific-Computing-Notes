\subsection{Iterative Methods for Linear Systems}
\label{sub:iterative-basics}

\textBox{%
A sparse matrix can have a factorization that is too expensive to store. Iterative methods instead improve an approximate solution through matrix--vector products and simpler solves. Their cost depends both on the cost of one correction and on how many corrections are needed.
}

\textBox{%
For an approximation $\bx$, compute the residual $\br=\bb-\bA\bx$. The exact correction satisfies $\bA\boldsymbol\delta=\br$. Solving that equation exactly would be the original task again. We instead solve $\bM\boldsymbol\delta=\br$ with a matrix $\bM$ that is cheap to use, and add the correction to $\bx$.
}

\subsubsection{Matrix Splittings and Stationary Iterations}

\textBox{%
The correction update is $\bx^{(k+1)}=\bx^{(k)}+\bM^{-1}(\bb-\bA\bx^{(k)})$. The inverse here denotes a solve with $\bM$, rather than a matrix to form. Rearranging this expression gives the traditional splitting $\bA=\bM-\mathbf N$ and the recurrence below. The same $\bM$ is used at every step, which explains the word \emph{stationary}.
}

\addDef{Stationary Iteration}{%
Given a splitting $\bA = \bM - \mathbf{N}$, the corresponding stationary iteration computes successive approximations via the fixed-point recurrence
\begin{align*}
    \bM\bx^{(k+1)} = \mathbf{N}\bx^{(k)} + \bb \iff \bx^{(k+1)} = \bT\bx^{(k)} + \bc,
\end{align*}
where $\bT = \bM^{-1}\mathbf{N} = \bI - \bM^{-1}\bA$ is the \textbf{iteration matrix} and $\bc = \bM^{-1}\bb$ is the modified right-hand side vector.
}
\label{def:stationary-iteration}

\addExcr{%
Consider the matrix $\bA = \begin{pmatrix} 4 & 1 \\ 2 & 3 \end{pmatrix}$ and the splitting $\bM = 4\bI$. Compute the iteration matrix $\bT = \bI - \bM^{-1}\bA$, find its eigenvalues, and evaluate its spectral radius $\rho(\bT)$.
}



\addRemark{%
The exact solution is a fixed point because its residual is zero. That fact establishes consistency of the iteration, but does not imply convergence to the fixed point. The correction must reduce every possible error direction. The following spectral-radius criterion states exactly when this occurs.
}

\subsubsection{Convergence Theory and the Spectral Radius}

\textBox{%
The convergence of a linear stationary iteration is determined entirely by the spectral properties of the iteration matrix $\bT$.
}

\addExcr{%
For the scalar equation $ax=b$ with $a>0$, use the correction $\delta=(b-ax)/m$ with $m>0$. Derive the error relation $e^{(k+1)}=(1-a/m)e^{(k)}$. Determine the values of $a/m$ for which the errors decrease, alternate in sign, remain unchanged in magnitude, or grow. The matrix criterion extends this calculation to all eigenvector directions.
}

\addDef{Spectral Radius}{%
Let $\bT \in \fR^{n \times n}$ with spectrum $\sigma(\bT) \subset \fC$. The spectral radius of $\bT$ is defined as the maximum modulus among all its eigenvalues:
\begin{align*}
    \rho(\bT) = \max \{ |\lambda| : \lambda \in \sigma(\bT) \}.
\end{align*}
}
\label{def:spectral-radius}

\addThm{Gelfand's Spectral Radius Theorem}{%
For any matrix norm $\|\cdot\|$ induced by a vector norm on $\fC^n$ and any matrix $\bT \in \fR^{n \times n}$,
\begin{align*}
    \rho(\bT) = \lim_{k \to \infty} \|\bT^k\|^{1/k}.
\end{align*}
In particular, for any $\varepsilon > 0$, there exists an induced matrix norm $\|\cdot\|_*$ such that $\|\bT\|_* \leq \rho(\bT) + \varepsilon$.
}
\label{thm:gelfand}

\addExcr{%
The following construction proves the norm assertion and the limit. Use Schur form $\bT=\bQ\bU\bQ^*$ and set $\bD=\operatorname{diag}(1,\eta,\ldots,\eta^{n-1})$. Show that every strictly upper triangular entry of $\bD^{-1}\bU\bD$ tends to zero as $\eta\to0$. Choose $\eta$ so that its infinity norm is at most $\rho(\bT)+\varepsilon$, and define $\|\bx\|_*=\|\bD^{-1}\bQ^*\bx\|_\infty$. Finally use equivalence of norms to obtain $\limsup_k\|\bT^k\|^{1/k}\leq\rho(\bT)+\varepsilon$ in any fixed norm, and an eigenvector to obtain the reverse bound. Let $\varepsilon\downarrow0$.
}


\addThm{Fundamental Convergence Criterion for Stationary Iterations}{%
The stationary iteration $\bx^{(k+1)} = \bT\bx^{(k)} + \bc$ converges to the unique solution $\bx^* = \bA^{-1}\bb$ for every initial guess $\bx^{(0)} \in \fR^n$ if and only if
\begin{align*}
    \rho(\bT) < 1.
\end{align*}
}
\label{thm:iteration-convergence}

\addPrf{%
Let $\be^{(k)} = \bx^{(k)} - \bx^*$ denote the error vector at step $k$. Subtracting the fixed-point identity $\bx^* = \bT\bx^* + \bc$ from the iteration formula $\bx^{(k+1)} = \bT\bx^{(k)} + \bc$:
\begin{align*}
    \be^{(k+1)}
    &= \bx^{(k+1)} - \bx^* \\
    &= (\bT\bx^{(k)} + \bc) - (\bT\bx^* + \bc) \\
    &= \bT(\bx^{(k)} - \bx^*)
    = \bT \be^{(k)}.
\end{align*}
Applying this recurrence recursively yields
\begin{align*}
    \be^{(k)} = \bT^k \be^{(0)}.
\end{align*}
The recurrence shows why the error, rather than the iterate itself, is the convenient object: the fixed right-hand side has cancelled. In particular, testing all initial guesses is equivalent to testing all initial errors, since any error vector can be realized by choosing $\bx^{(0)}=\bx^*+\be^{(0)}$.

The error $\be^{(k)} \to \bzero$ for all initial errors $\be^{(0)} \in \fR^n$ if and only if $\lim_{k\to\infty} \bT^k = \bzero$.

If $\rho(\bT) < 1$, choose $\varepsilon = \frac{1 - \rho(\bT)}{2} > 0$. By Gelfand's theorem (Thm~\ref{thm:gelfand}), there exists an induced matrix norm $\|\cdot\|_*$ such that
\begin{align*}
    \|\bT\|_* \leq \rho(\bT) + \varepsilon = \frac{1 + \rho(\bT)}{2} < 1.
\end{align*}
Then $\|\bT^k\|_* \leq \|\bT\|_*^k \to 0$ as $k \to \infty$, so $\bT^k \to \bzero$ and $\|\be^{(k)}\|_* \leq \|\bT\|_*^k \|\be^{(0)}\|_* \to 0$.

Conversely, suppose $\bT^k \to \bzero$. If $\lambda \in \sigma(\bT)$ with eigenvector $\bv \neq \bzero$, then $\bT^k \bv = \lambda^k \bv$. For this sequence to converge to $\bzero$, we must have $|\lambda|^k \to 0$, forcing $|\lambda| < 1$. If an eigenvector is complex while the matrix is real, write it as $\bv=\ba+i\bb$. Convergence of $\bT^k$ on every real vector makes both $\bT^k\ba$ and $\bT^k\bb$ tend to zero, so convergence also holds for their complex combination. Thus the eigenvector test covers complex eigenvalues as well.

Because this holds for all eigenvalues, $\rho(\bT) < 1$.
}

\addRemark{%
\textbf{(Asymptotic convergence rate and iteration count)}\enspace
The quantity $-\log_{10} \rho(\bT)$ represents the asymptotic rate of convergence, measuring the number of decimal digits of accuracy gained per iteration. To reduce the initial error by a relative tolerance factor $\varepsilon$, the required number of iterations satisfies
\begin{align*}
    k \approx \frac{\ln(1/\varepsilon)}{-\ln \rho(\bT)}.
\end{align*}
If $\rho(\bT) = 0.99$, we have $-\ln(0.99) \approx 0.01005$, which requires approximately $230$ iterations to reduce the error by a factor of ten. As $\rho(\bT) \to 1$, convergence slows drastically.
}

\subsubsection{Krylov Subspaces and Matrix-Free Solvers}

\textBox{%
A stationary method applies the same splitting at every step. A Krylov method instead chooses the iterate $\bx^{(k)}$ from the affine space $\bx^{(0)}+\mathcal{K}_k(\bA,\br^{(0)})$, where $\mathcal{K}_k$ is the Krylov subspace of Def~\ref{def:krylov-subspace} generated by the initial residual. The basis for this space is built by the Arnoldi process of Section~\ref{sub:arnoldi}, or by the Lanczos recurrence when $\bA$ is symmetric. The conjugate gradient and GMRES methods differ in the optimality criterion used to select $\bx^{(k)}$.
}

\addThm{Krylov Polynomial Representation}{%
Let $\bx^{(0)}$ be an initial guess with residual $\br^{(0)} = \bb - \bA\bx^{(0)}$. Any iterate $\bx^{(k)}$ belonging to the affine Krylov space $\bx^{(0)} + \mathcal{K}_k(\bA, \br^{(0)})$ can be represented as
\begin{align*}
    \bx^{(k)} = \bx^{(0)} + p_{k-1}(\bA)\br^{(0)},
\end{align*}
where $p_{k-1}$ is a polynomial of degree at most $k-1$. The corresponding residual vector satisfies
\begin{align*}
    \br^{(k)} = \bb - \bA\bx^{(k)} = q_k(\bA)\br^{(0)},
\end{align*}
where $q_k(t) = 1 - t p_{k-1}(t)$ is a polynomial of degree at most $k$ satisfying the normalization constraint $q_k(0) = 1$.
}
\label{thm:krylov-polynomial}

\addExcr{%
Write a general correction as $\sum_{j=0}^{k-1}c_j\bA^j\br^{(0)}$ and subtract its image from $\br^{(0)}$. Prove the polynomial representation. Conversely, if $q_k(0)=1$, show that $(1-q_k(t))/t$ is a polynomial of degree at most $k-1$.
}


\addRemark{%
\textbf{(Matrix-free computation)}\enspace
Krylov subspace methods never require explicit access to the individual entries of $\bA$. They interact with the linear operator exclusively by computing matrix--vector products $\bv \mapsto \bA\bv$. Consequently, linear systems can be solved efficiently even when the matrix is defined implicitly as a differential operator or a numerical simulation routine.
}

\addDef{Stopping Criteria for Iterative Solvers}{%
Because the true error $\be^{(k)} = \bx^{(k)} - \bx^*$ is unobservable, termination is determined using the computable residual vector $\br^{(k)} = \bb - \bA\bx^{(k)}$. Common stopping criteria include:
\begin{enumerate}
    \item \textbf{Relative residual test:} $\frac{\|\br^{(k)}\|_2}{\|\bb\|_2} \leq \texttt{tol}$.
    \item \textbf{Preconditioned relative residual test:} $\frac{\|\bM^{-1}\br^{(k)}\|_2}{\|\bM^{-1}\bb\|_2} \leq \texttt{tol}$.
    \item \textbf{Step difference test:} $\frac{\|\bx^{(k+1)} - \bx^{(k)}\|_2}{\|\bx^{(k+1)}\|_2} \leq \texttt{tol}$.
\end{enumerate}
}
\label{def:stopping-criteria}

\addExcr{%
Prove that if $\|\bT\| < 1$ for some induced matrix norm, then the error satisfies $\|\be^{(k)}\| \leq \frac{\|\bT\|}{1 - \|\bT\|} \|\bx^{(k)} - \bx^{(k-1)}\|$.
}


\textBox{%
Relative tests require a nonzero denominator; for $\bb=\bzero$, use an absolute residual tolerance. A small step alone can also signal stagnation. Recompute the true residual before accepting convergence, and use the residual--error bound to interpret its size. The normwise backward-error denominator $\|\bA\|\|\bx^{(k)}\|+\|\bb\|$ often gives a more useful scale than $\|\bb\|$ alone.
}
