\subsection{The Lanczos Algorithm}
\label{sub:lanczos-theory}

\textBox{%
When $\bA$ is real symmetric, the Arnoldi process of Section~\ref{sub:arnoldi} simplifies in two ways. First, the projected matrix $\bH_k=\bV_k^T\bA\bV_k$ is symmetric as well as Hessenberg, so it is tridiagonal. Each new basis vector then needs to be orthogonalized only against the two preceding ones, and the cost per step no longer grows with $k$. Second, the Rayleigh quotient characterization of Thm~\ref{thm:courant-fischer} applies to the projected matrix. The characterization shows that the extreme Ritz values are the best estimates of the extreme eigenvalues available from the Krylov subspace.
}

\textBox{%
We first derive the three-term recurrence that defines the Lanczos algorithm \citep{lanczos1950}. We then show that the extreme Ritz values move monotonically toward the extreme eigenvalues and that they are never worse than the power-iteration estimate obtained with the same number of products. A numerical example compares the two methods. The section ends with the loss of orthogonality that occurs in floating-point arithmetic.
}

\subsubsection{Symmetry Gives a Three-Term Recurrence}

\addThm{Tridiagonal Projection}{%
Let $\bA\in\fR^{n\times n}$ be symmetric, and suppose the Arnoldi iteration (Def~\ref{def:arnoldi-iteration}) completes $k$ steps. Then $\bH_k$ is symmetric tridiagonal. Writing $\alpha_j=h_{jj}$ and $\beta_j=h_{j+1,j}$, we have
\begin{align*}
    \bH_k=\bT_k=\begin{pmatrix}
        \alpha_1 & \beta_1 & & \\
        \beta_1 & \alpha_2 & \ddots & \\
        & \ddots & \ddots & \beta_{k-1} \\
        & & \beta_{k-1} & \alpha_k
    \end{pmatrix},
\end{align*}
and the Arnoldi coefficients satisfy $h_{ij}=0$ for $i<j-1$ and $h_{j-1,j}=\beta_{j-1}$.
}
\label{thm:lanczos-tridiagonal}

\addPrf{%
By Thm~\ref{thm:arnoldi-factorization}, $\bH_k=\bV_k^T\bA\bV_k$. Since $\bA^T=\bA$,
\begin{align*}
    \bH_k^T=\bV_k^T\bA^T\bV_k=\bV_k^T\bA\bV_k=\bH_k.
\end{align*}
The matrix $\bH_k$ is upper Hessenberg, so $h_{ij}=0$ for $i>j+1$. Symmetry gives $h_{ij}=h_{ji}=0$ for $i<j-1$ as well, which leaves only the diagonal and the two adjacent diagonals. Symmetry also gives $h_{j-1,j}=h_{j,j-1}=\beta_{j-1}$.
}

\textBox{%
In the Arnoldi iteration, the coefficient $h_{ij}$ is the component of $\bA\bv_j$ along $\bv_i$. Theorem~\ref{thm:lanczos-tridiagonal} shows that these components vanish for $i<j-1$ in exact arithmetic, so the inner loop of Step 2 reduces to two subtractions. The coefficient for $i=j-1$ is already known from the previous step. Substituting these facts into Def~\ref{def:arnoldi-iteration} gives the Lanczos iteration.
}

\addDef{The Lanczos Iteration}{%
Let $\bA\in\fR^{n\times n}$ be symmetric and let $\bv_1$ be a unit vector. Set $\bv_0=\bzero$ and $\beta_0=0$. For $j=1,2,\dots,k$:
\begin{enumerate}
    \item Compute $\bw=\bA\bv_j-\beta_{j-1}\bv_{j-1}$.
    \item Compute $\alpha_j=\bv_j^T\bw$ and update $\bw\leftarrow\bw-\alpha_j\bv_j$.
    \item Compute $\beta_j=\|\bw\|_2$.
    \item If $\beta_j=0$, stop. Otherwise, set $\bv_{j+1}=\bw/\beta_j$.
\end{enumerate}
}
\label{def:lanczos-iteration}

\addCor{The Lanczos Relation}{%
After $k$ steps of the Lanczos iteration without breakdown, the matrix $\bV_k=(\bv_1\ \cdots\ \bv_k)$ has orthonormal columns spanning $\mathcal{K}_k(\bA,\bv_1)$, and
\begin{align*}
    \bA\bV_k=\bV_k\bT_k+\beta_k\bv_{k+1}\be_k^T,
    \qquad
    \bT_k=\bV_k^T\bA\bV_k.
\end{align*}
}
\label{thm:lanczos-relation}

\addPrf{%
In exact arithmetic the Lanczos iteration computes the same vectors and coefficients as the Arnoldi iteration, because it omits only the coefficients that Thm~\ref{thm:lanczos-tridiagonal} shows to be zero. The claims are therefore Thm~\ref{thm:arnoldi-factorization} with $\bH_k=\bT_k$ and $h_{k+1,k}=\beta_k$.
}

\addRemark{%
\textbf{(Lanczos cost)}\enspace
Each step requires one product with $\bA$ and $O(n)$ further operations, independent of $k$. If only eigenvalues are wanted, the iteration needs to keep just $\bv_{j-1}$, $\bv_j$, and the scalars $\alpha_j,\beta_j$. If Ritz vectors are also wanted, we either store $\bV_k$ or run the recurrence a second time to form $\bV_k\bs$.
}

\subsubsection{Ritz Values as Optimal Estimates}

\textBox{%
The Ritz pairs of Def~\ref{def:ritz-pair} are now the eigenpairs of the symmetric tridiagonal matrix $\bT_k$, so the Ritz values are real. Theorem~\ref{thm:ritz-residual} gives their residuals as $\beta_k|s_k|$. Symmetry also lets us say how good the Ritz values are. Throughout, let $\lambda_1\leq\cdots\leq\lambda_n$ be the eigenvalues of $\bA$ and let $\theta_1^{(k)}\leq\cdots\leq\theta_k^{(k)}$ be the eigenvalues of $\bT_k$.
}

\addThm{Rayleigh--Ritz Optimality}{%
Let $R(\bx)=\bx^T\bA\bx/\bx^T\bx$. The extreme Ritz values are the extreme values of the Rayleigh quotient over the Krylov subspace,
\begin{align*}
    \theta_k^{(k)}=\max_{\bx\in\mathcal{K}_k,\,\bx\neq\bzero}R(\bx),
    \qquad
    \theta_1^{(k)}=\min_{\bx\in\mathcal{K}_k,\,\bx\neq\bzero}R(\bx).
\end{align*}
Consequently, $\lambda_1\leq\theta_1^{(k)}$ and $\theta_k^{(k)}\leq\lambda_n$.
}
\label{thm:rayleigh-ritz-optimality}

\addPrf{%
Every nonzero $\bx\in\mathcal{K}_k$ has the form $\bx=\bV_k\by$ with $\by\in\fR^k$ nonzero. Orthonormality of the columns of $\bV_k$ gives $\bx^T\bx=\by^T\by$, and Cor~\ref{thm:lanczos-relation} gives $\bx^T\bA\bx=\by^T\bT_k\by$. Hence
\begin{align*}
    R(\bx)=\frac{\by^T\bT_k\by}{\by^T\by}.
\end{align*}
Applying Thm~\ref{thm:courant-fischer} to the symmetric matrix $\bT_k$ with $k=1$ and with the largest index shows that the minimum and maximum of this quotient over $\by\neq\bzero$ are $\theta_1^{(k)}$ and $\theta_k^{(k)}$. Applied to $\bA$, the same characterization shows that $R(\bx)$ lies in $[\lambda_1,\lambda_n]$ for every nonzero $\bx\in\fR^n$, which gives the final inequalities.
}

\addCor{Monotone Convergence of the Extreme Ritz Values}{%
The Ritz values at consecutive steps interlace,
\begin{align*}
    \theta_i^{(k+1)}\leq\theta_i^{(k)}\leq\theta_{i+1}^{(k+1)},
    \qquad i=1,\dots,k.
\end{align*}
In particular, $\theta_k^{(k)}$ increases toward $\lambda_n$ and $\theta_1^{(k)}$ decreases toward $\lambda_1$ as $k$ grows.
}
\label{cor:ritz-interlacing}

\addPrf{%
The matrix $\bT_k$ is obtained from $\bT_{k+1}$ by deleting its last row and column. Applying Thm~\ref{thm:cauchy-interlacing} with $\bT_{k+1}$ in place of $\bA$ and $\bT_k$ in place of $\bB$ gives the interlacing inequalities. Taking $i=k$ in the right inequality gives $\theta_k^{(k)}\leq\theta_{k+1}^{(k+1)}$, and taking $i=1$ in the left inequality gives $\theta_1^{(k+1)}\leq\theta_1^{(k)}$. Theorem~\ref{thm:rayleigh-ritz-optimality} bounds the two sequences by $\lambda_n$ and $\lambda_1$.
}

\addCor{Lanczos Improves on Power Iteration}{%
Let $\bA$ be symmetric positive definite, and let $\lambda^{(k-1)}$ be the Rayleigh quotient estimate of Def~\ref{def:power-iteration} after $k-1$ power steps from $\bv^{(0)}=\bv_1$. Both $\lambda^{(k-1)}$ and $\theta_k^{(k)}$ require $k$ products with $\bA$, and
\begin{align*}
    \lambda^{(k-1)}\leq\theta_k^{(k)}\leq\lambda_n.
\end{align*}
}
\label{cor:lanczos-vs-power}

\addPrf{%
The power iterate $\bv^{(k-1)}$ is a multiple of $\bA^{k-1}\bv_1$, so it lies in $\mathcal{K}_k(\bA,\bv_1)$. Its Rayleigh quotient is $\lambda^{(k-1)}=R(\bv^{(k-1)})$, which is at most the maximum in Thm~\ref{thm:rayleigh-ritz-optimality}. Power iteration uses $k-1$ products to form $\bv^{(k-1)}$ and one more to evaluate its Rayleigh quotient. Lanczos uses one product per step. Positive definiteness makes $\lambda_n$ the eigenvalue of largest magnitude, so the two methods approximate the same eigenvalue.
}

\textBox{%
The polynomial form of Def~\ref{def:krylov-subspace} explains the size of the improvement. Every vector in $\mathcal{K}_k$ is $p(\bA)\bv_1$ for some $p\in\mathcal{P}_{k-1}$. Expanding $\bv_1=\sum_i c_i\bq_i$ in the eigenvectors of $\bA$ gives
\begin{align*}
    \theta_k^{(k)}=\max_{p\in\mathcal{P}_{k-1}}R(p(\bA)\bv_1)
    =\max_{p\in\mathcal{P}_{k-1}}\frac{\sum_{i=1}^n\lambda_i\,p(\lambda_i)^2c_i^2}{\sum_{i=1}^n p(\lambda_i)^2c_i^2}.
\end{align*}
The quotient is close to $\lambda_n$ when $p$ is large at $\lambda_n$ and small on the rest of the spectrum. Power iteration is restricted to $p(t)=t^{k-1}$, which separates $\lambda_n$ from $\lambda_{n-1}$ only by the factor $(\lambda_{n-1}/\lambda_n)^{k-1}$. Lanczos implicitly selects the best polynomial. A shifted Chebyshev polynomial, which is bounded by one on $[\lambda_1,\lambda_{n-1}]$ and grows rapidly outside it, gives the following bound.
}

\addRemark{%
\textbf{(Kaniel--Paige bound)}\enspace
Let $\phi$ be the angle between $\bv_1$ and the eigenvector $\bq_n$, let $\gamma=(\lambda_n-\lambda_{n-1})/(\lambda_{n-1}-\lambda_1)$, and let $T_m(x)=\cosh(m\operatorname{arccosh}x)$ for $x\geq1$. Then
\begin{align*}
    0\leq\lambda_n-\theta_k^{(k)}\leq(\lambda_n-\lambda_1)\left(\frac{\tan\phi}{T_{k-1}(1+2\gamma)}\right)^2.
\end{align*}
A proof is given by Golub and Van Loan \citep{golubvanloan2013}. Since $T_m(1+2\gamma)$ grows roughly like $(1+2\sqrt{\gamma})^m$ when $\gamma$ is small, the error decreases geometrically at a rate set by the square root of the relative gap. A square root of the same kind appears in the conjugate gradient bound of Thm~\ref{thm:cg-convergence-bound}. The analogous bound for $\theta_1^{(k)}$ follows by applying the result to $-\bA$. Interior eigenvalues admit weaker bounds and converge more slowly, because no low-degree polynomial can be large at an interior point and small on both sides of it.
}

\addExmpl{%
\textbf{(Lanczos and power iteration on a model spectrum)}\enspace
Let $\bA=\operatorname{diag}(1,2,\dots,100)$ and $\bv_1=(1,\dots,1)^T/10$. We run Lanczos with full reorthogonalization, so that the computed Ritz values agree with exact arithmetic to rounding level, and compare the error in the largest Ritz value with the error in the power-iteration estimate after the same number of products.
\begin{center}
\begin{tabular}{rccc}
    $k$ & $\lambda_n-\theta_k^{(k)}$ & $\lambda_n-\lambda^{(k-1)}$ & Kaniel--Paige bound \\
    \hline
    10 & $8.6\times10^{-1}$ & $4.5$ & $9.9\times10^{2}$ \\
    20 & $5.0\times10^{-2}$ & $2.0$ & $1.8\times10^{1}$ \\
    30 & $6.4\times10^{-4}$ & $1.2$ & $3.3\times10^{-1}$ \\
    40 & $7.2\times10^{-7}$ & $8.2\times10^{-1}$ & $5.8\times10^{-3}$
\end{tabular}
\end{center}
The relative gap is $\gamma=1/98$, and the power-iteration ratio is $\lambda_{99}/\lambda_{100}=0.99$. After 40 products the power-iteration estimate has not yet reached one correct digit, whereas the largest Ritz value is correct to about eight significant digits. The bound is valid but conservative at every step. By the symmetry of this spectrum about $50.5$ and of the starting vector, the error in $\theta_1^{(k)}$ is the same as the error in $\theta_k^{(k)}$.

At $k=40$ the residual of Thm~\ref{thm:ritz-residual} for the largest Ritz pair is $3.7\times10^{-3}$, about four orders of magnitude larger than the eigenvalue error. Corollary~\ref{cor:symmetric-residual} explains the difference. The corollary bounds the distance from the Ritz value to the nearest eigenvalue by the squared residual divided by the distance $\delta$ from the Ritz value to the other eigenvalues. Here $\delta$ is about one, and the bound $1.4\times10^{-5}$ is consistent with the observed error.
}

\addRemark{%
\textbf{(Lanczos and conjugate gradients)}\enspace
The conjugate gradient method of Section~\ref{sub:cg} builds the same Krylov subspace $\mathcal{K}_k(\bA,\br^{(0)})$ for a symmetric positive definite system. Its residuals are multiples of the Lanczos vectors started from $\bv_1=\br^{(0)}/\|\br^{(0)}\|_2$, and its coefficients determine the entries of $\bT_k$. Section~\ref{sub:cg} derives this correspondence and uses it to estimate the extreme eigenvalues during a CG solve.
}

\subsubsection{Loss of Orthogonality and Ghost Eigenvalues}

\textBox{%
Theorem~\ref{thm:lanczos-tridiagonal} relies on the exact orthogonality of $\bv_{j+1}$ to $\bv_1,\dots,\bv_{j-2}$. The Lanczos iteration never enforces this orthogonality explicitly, so rounding errors are free to destroy it.
}

\addRemark{%
\textbf{(Paige's analysis)}\enspace
In floating-point arithmetic, the computed Lanczos vectors gradually lose mutual orthogonality. Paige showed that the loss is not random \citep{paige1976}. Let $(\theta_i,\tilde{\bu}_i)$ be a Ritz pair computed from $\bT_k$, and let $u$ denote the unit roundoff. Paige's analysis gives
\begin{align*}
    |\tilde{\bu}_i^T\bv_{k+1}|\approx\frac{u\,\|\bA\|_2}{\beta_k|s_{k,i}|}.
\end{align*}
The denominator is the residual of Thm~\ref{thm:ritz-residual}. The new Lanczos vector therefore acquires a component along $\tilde{\bu}_i$ precisely as that Ritz pair converges, and orthogonality is lost in the directions of converged Ritz vectors. Once orthogonality to a converged Ritz vector is lost, the iteration begins to approximate the same eigenvalue again and produces repeated copies of it, known as \textbf{ghost eigenvalues}. In the example above, running the iteration without reorthogonalization for $k=150$ steps gives three Ritz values within $10^{-6}$ of $\lambda_{100}=100$ and three within $10^{-6}$ of $\lambda_1=1$. At $k=200$ there are four copies of each.
}

\addRemark{%
\textbf{(Remedies for ghost eigenvalues)}\enspace
Practical implementations use one of three strategies.
\begin{enumerate}
    \item \textbf{Full reorthogonalization.} We orthogonalize each new vector against all previous Lanczos vectors. This restores the behavior of exact arithmetic at the Arnoldi cost of $O(k^2n)$ operations and $O(kn)$ storage.
    \item \textbf{Selective reorthogonalization.} We orthogonalize only against converged Ritz vectors, and only when Paige's analysis indicates that orthogonality to them has been lost. The cost is proportional to the number of converged Ritz vectors.
    \item \textbf{Cullum--Willoughby identification.} We run the iteration without reorthogonalization and discard spurious Ritz values afterward by comparing the eigenvalues of $\bT_k$ with those of the matrix obtained by deleting its first row and column.
\end{enumerate}
}
